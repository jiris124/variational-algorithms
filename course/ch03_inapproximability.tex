\chapter{Inapproximability and the Unique Games Conjecture}\label{C.Inapprox}

\ona{In \Chap~\chref{C.Motivation} we saw that the Goemans--Williamson algorithm cuts, in expectation, at least $\alpha_{\rm GW} \approx 0.878$ of the edges that an optimal cut would. In this \chap{} we ask whether one can do better in polynomial time, by any algorithm, classical or quantum. The answer depends on a conjecture, the Unique Games Conjecture of \citet{khot2002power}, which we state in several equivalent forms, and on a delicate piece of analysis showing that, if it holds, the constant $\alpha_{\rm GW}$ cannot be improved. In September 2026 a preprint \citep{OAI:The-Unique-Games-Theorem-September-23-2026} claimed a proof of the conjecture; it has not yet been peer reviewed, and we describe what it claims and what would follow from it. The hardness of training a variational circuit, as opposed to the quality of the solutions it can reach, is the subject of \Chap~\chref{C.VQE}. Our sources are the survey \citep{abbas2024challenges}, the appendices of \citet{egger2021warm}, and the survey of \citet{KhotSurvey}.}

\section{Complexity classes for optimization}

\begin{frame}\ft{Decision, function, optimization, approximation}
\ona{Decision problems are classified by $\mathbf{P}$ and $\mathbf{NP}$; relational (function) problems by $\mathbf{FP}$ and $\mathbf{FNP}$; optimization problems by $\mathbf{PO} \subset \mathbf{FP}$ and $\mathbf{NPO} \subset \mathbf{FNP}$. For a maximisation problem, an algorithm has \emph{approximation ratio} $\alpha \le 1$ if on every instance it returns a solution of value at least $\alpha\cdot\mathrm{OPT}$; for randomised algorithms one asks this in expectation. The classes of approximable problems refine the picture, as in Fig.~\ref{fig:ch03:classes}:}
\begin{itemize}\setlength\itemsep{0.2em}
    \item $\mathbf{APX}$: $\mathbf{NPO}$ problems with a polynomial-time constant-factor approximation (MaxCut, MAX-3-SAT, metric TSP).
    \item $\mathbf{PTAS}$: a $(1+\epsilon)$-approximation in polynomial time for every fixed $\epsilon>0$ (Euclidean TSP).
    \item $\mathbf{FPTAS}$: the running time is polynomial in $1/\epsilon$ as well (Knapsack).
\end{itemize}
\onp{\begin{center}\includegraphics[height=0.4\textheight]{figures/survey/optimization_classes.pdf}\end{center}\vspace{-0.6em}{\tiny Complexity classes for optimization problems \citep{abbas2024challenges}.}}
\ona{\begin{figure}[htb]\centering\includegraphics[width=0.644\linewidth,height=0.294\textheight,keepaspectratio]{figures/survey/optimization_classes.pdf}\caption{Complexity classes for optimization problems \citep{abbas2024challenges}.}\label{fig:ch03:classes}\end{figure}}
\end{frame}

\begin{frame}\ft{MaxCut and QUBO}
\begin{itemize}\setlength\itemsep{0.3em}
    \item The decision version of MaxCut is $\mathbf{NP}$-complete \citep{Garey1976}; the optimization version is $\mathbf{APX}$-hard \citep{lee2021classifying}. It has a constant-factor approximation, but no PTAS unless $\mathbf P = \mathbf{NP}$.
    \item QUBO, $\min_{x\in\{0,1\}^n}x^\top Qx$, contains MaxCut as a special case; it is $\mathbf{NP}$-hard and $\mathbf{APX}$-hard. For its continuous relaxation over the box $[0,1]^n$, even deciding whether a given point is a local minimum is $\mathbf{NP}$-hard \citep{murty1987some}.
    \item The substitution $x_i \mapsto (1-Z_i)/2$ turns QUBO into a diagonal Hamiltonian whose ground states are the optima. The ground-state problem for diagonal two-local Hamiltonians is therefore $\mathbf{NP}$-hard, and that for general local Hamiltonians is $\mathbf{QMA}$-hard (\Chap~\chref{C.EQA}).
\end{itemize}
\ona{No quantum algorithm is expected to solve MaxCut exactly in polynomial time, then, and the interesting question is how well it can be approximated.}
\end{frame}

\ona{Figure~\ref{fig:ch03:line} collects the constants that will occupy us. Assigning each vertex to a side at random cuts half the edges in expectation, so $1/2$ is trivial. It is $\mathbf{NP}$-hard to do better than $16/17 \approx 0.9412$ \citep{Hstad2001}, and, under the Unique Games Conjecture, already to do better than $\alpha_{\rm GW}$ \citep{khot2007optimal}; if the claimed proof of the conjecture \citep{OAI:The-Unique-Games-Theorem-September-23-2026} stands, the latter is $\mathbf{NP}$-hard unconditionally. When negative weights are allowed the relevant constant is $11/13 \approx 0.846$, as we discuss below; and on 3-regular graphs the picture is different again (Fig.~\ref{fig:ch03:cubic}).}

\begin{frame}\ft{MaxCut on 3-regular graphs}
\ona{Restricting the input does not make MaxCut easy. Deleting the fewest edges so that the remaining graph is bipartite is the same as keeping the most edges in a cut, and \citet{yannakakis1978node} studied this ``edge-deletion bipartite'' problem among many edge-deletion problems, determining ``the best possible bounds on the node-degrees for which the problems remain NP-complete''. For MaxCut the bound is three. With maximum degree two the graph is a disjoint union of paths and cycles, and a maximum cut keeps every edge except one on each odd cycle; with degree three the problem is already hard:}
\begin{thm}[\citet{yannakakis1978node}]\label{thm:ch03:cubic}
The decision version of MaxCut is $\mathbf{NP}$-complete on 3-regular (cubic) graphs.
\end{thm}
\ona{On these graphs the constants are different from those of Fig.~\ref{fig:ch03:line}, see Fig.~\ref{fig:ch03:cubic}. Semidefinite programming does better than in general: \citet{halperin2004max} achieve $0.9326$ for graphs of maximum degree three. It is $\mathbf{NP}$-hard to approximate MaxCut on 3-regular graphs within $0.997$ \citep{berman1999some}, so a small gap remains. The QAOA guarantees are on this class of graphs: $0.6924$ at depth one \citep{farhi2014quantum}, $0.7559$ at depth two and, under assumptions, $0.7924$ at depth three \citep{PhysRevA.103.042612}. All lie below the Goemans--Williamson guarantee, which holds for every graph.}

\begin{frame}\ft{Where the hardness on cubic graphs comes from}
\onp{\footnotesize}
\ona{MaxCut itself was shown $\mathbf{NP}$-complete by \citet{Garey1976}, who left open its status on graphs of bounded degree; the weighted version goes back to \citet{karp1972reducibility}. \citet{yannakakis1978node} settles the question for cubic graphs in his Theorem~13, whose proof in the paper consists of one sentence: it ``uses Lemmas 2 and 5 (rather the transformations employed there)''. The two lemmas concern 3-SAT, the satisfiability of formulae in which each clause is a disjunction of three literals:}
\begin{itemize}\setlength\itemsep{0.3em}
    \item \textbf{Lemma 5.} 3-SAT remains $\mathbf{NP}$-complete when each variable appears three times. This is best possible: if each variable appears at most twice, and each clause has at least two literals, the formula is always satisfiable.
    \item \textbf{Lemma 2.} A 3-SAT formula $S$ can be transformed in polynomial time into a formula $S'$, with linearly many clauses of at most three literals, such that: $S'$ is satisfiable if and only if $S$ is; every variable whose complement occurs in $S'$ occurs as often as its complement; if $S$ is satisfiable, every satisfying assignment of $S'$ makes at most two literals true in each clause, and some satisfying assignment sets all uncomplemented variables true; and if $S$ is not satisfiable, every assignment setting all uncomplemented variables true makes all three literals true in some clause.
\end{itemize}
\ona{The paper gives no more detail, but the role of the two lemmas can be explained. A cut is a $\pm1$ assignment to the vertices, as a truth assignment is to the variables, and a triangle is the natural test of a clause: a cut separates at most two of its three edges, and exactly two unless all three vertices lie on the same side. A clause gadget built from triangles therefore rewards assignments that do not make all literals of a clause equal, and Lemma~2 puts satisfiability in a form, ``at most two literals true in each clause'', that such gadgets can detect. Lemma~5 bounds the number of clauses in which each variable occurs, so that each vertex of the graph needs only a bounded number of edges; at most three are allowed in the end. Three is also the best possible bound, since with maximum degree two the problem is easy.}
\onp{\scriptsize Proof in the paper: ``uses Lemmas 2 and 5''. Idea: a triangle has at most 2 of 3 edges cut, exactly 2 unless all its vertices lie on one side; Lemma 2 makes satisfiability a condition of this kind; Lemma 5 keeps the degrees bounded.}
\end{frame}

\section{Constraint satisfaction problems}

\begin{frame}\ft{Constraint satisfaction problems}
\ona{Most of the problems in this \chap{} share a common form. A \emph{constraint satisfaction problem} (CSP) consists of}
\begin{itemize}\setlength\itemsep{0.25em}
    \item variables $x_1,\dots,x_n$ taking values in a finite \emph{alphabet} $\Sigma$, say $\Sigma = \{0,1,\dots,k-1\}$;
    \item constraints $C_1,\dots,C_m$, each involving a few of the variables, its \emph{arity} being their number, and each specifying which combinations of their values are allowed.
\end{itemize}
\ona{The \emph{value} of an assignment $x \in \Sigma^n$ is the fraction of constraints it satisfies, and the value of the instance is the largest value of any assignment. Max-CSP asks for an assignment of largest value, and the approximation ratio compares the value found with the largest. Examples:}
\begin{center}\ona{\small}\onp{\scriptsize}
\begin{tabular}{llll}\toprule
problem & alphabet & arity & a constraint \\\midrule
Max-3-SAT & $\{0,1\}$ & $3$ & $x_1 \vee \bar x_2 \vee x_5$ \\
MaxCut & $\{0,1\}$ & $2$ & $x_u \ne x_v$, one for each edge $uv$ \\
Max-$k$-colouring & $\{0,\dots,k-1\}$ & $2$ & $x_u \ne x_v$ \\
Max-2-Lin$(q)$ & $\mathbb Z_q$ & $2$ & $x_u - x_v = c_{uv} \pmod q$ \\\bottomrule
\end{tabular}
\end{center}
\ona{The PCP theorem is a statement about CSPs: for Max-3-SAT it is $\mathbf{NP}$-hard to distinguish satisfiable instances from those of value at most some $s<1$.}
\onp{\footnotesize Value of an assignment $=$ fraction of constraints satisfied.}
\end{frame}

\begin{frame}\ft{Binary CSPs as labelled graphs}
\onp{\footnotesize}
\ona{When every constraint involves two variables (arity two), a CSP can be drawn as a graph: one vertex for each variable, one edge for each constraint. Following the terminology of PCPs, the values are called \emph{labels} and an assignment a \emph{labelling}. The constraint on an edge $uv$ is a relation $R_{uv} \subseteq \Sigma\times\Sigma$, the set of allowed pairs of labels, which we draw as lines between two columns of labels (Fig.~\ref{fig:ch03:constraints}). Three kinds of relation matter:}

\begin{itemize}\setlength\itemsep{0.2em}
    \item \textbf{Label cover:} a bipartite graph with \emph{projection} constraints, $x_v = \pi_e(x_w)$ for a map $\pi_e$. By the PCP theorem and parallel repetition, for every $\delta>0$ there is an alphabet size for which it is $\mathbf{NP}$-hard to distinguish label cover instances of value $1$ from those of value at most $\delta$; most strong inapproximability results start from this.
    \item \textbf{Unique label cover:} the same, but each $\pi_e$ is a \emph{bijection}. MaxCut, with the single bijection ``swap'' on $\{0,1\}$, and Max-2-Lin$(q)$, with rotations, are examples.
\end{itemize}
\onp{\footnotesize Label cover: projections; unique label cover: bijections.}
\end{frame}

\section{Hardness of approximation}

\begin{frame}\ft{Gap problems and the PCP theorem}
\ona{Statements of inapproximability are best phrased as \emph{gap problems}. For a constraint satisfaction problem (CSP) and numbers $0 < s < c \le 1$, the problem $\mathrm{Gap}_{c,s}$ asks us to decide, given an instance promised to have value either at least $c$ (the \emph{completeness}) or at most $s$ (the \emph{soundness}), which of the two holds. If $\mathrm{Gap}_{c,s}$ is $\mathbf{NP}$-hard, then no polynomial-time algorithm has approximation ratio better than $s/c$, unless $\mathbf P = \mathbf{NP}$: such an algorithm would separate the two cases.}
\begin{thm}[PCP theorem, gap form]
There is a constant $s<1$ such that $\mathrm{Gap}_{1,s}$ for MAX-3-SAT is $\mathbf{NP}$-hard.
\end{thm}
\ona{The PCP theorem thus says that the fraction of satisfiable constraints is hard to approximate within \emph{some} constant. Sharpening the constant for particular problems has occupied a large literature. For MaxCut the best unconditional result is:}
\begin{thm}[\citet{Hstad2001}]
For every $\epsilon > 0$, it is $\mathbf{NP}$-hard to approximate MaxCut with non-negative weights within $16/17 + \epsilon \approx 0.941$.
\end{thm}
\ona{Between $\alpha_{\rm GW} \approx 0.878$ and $0.941$ there remains a gap. The Unique Games Conjecture closes it, and a 2026 preprint claims to prove the conjecture (Section~\ref{sec:ch03:ugt}).}
\onp{\footnotesize $\mathrm{Gap}_{c,s}$ $\mathbf{NP}$-hard $\Rightarrow$ no ratio better than $s/c$ unless $\mathbf P = \mathbf{NP}$.}
\end{frame}

\section{Unique games}

\begin{frame}\ft{Unique label cover}
\ona{Formally, in a \emph{unique label cover} problem there is a bipartite graph $G = (V,W,E)$, an alphabet $M$ of $k$ labels, and for each edge $e$ a bijection $\pi_e : M \to M$. Given labellings $\mathcal A_V : V\to M$ and $\mathcal A_W : W \to M$, the edge $e = (v,w)$ is satisfied if $\pi_e(\mathcal A_W(w)) = \mathcal A_V(v)$, and $\opt(\mathcal L)$ is the largest fraction of edges that can be satisfied simultaneously. The word \emph{unique} refers to the bijections: the label of one endpoint determines the label of the other. ULC$(\delta)$ asks us to decide whether $\opt(\mathcal L) \ge 1-\delta$ or $\opt(\mathcal L) \le \delta$.}
\figslide[height=0.38\textheight]{figures/warmstart/figure11.pdf}{Unique label cover \citep{egger2021warm}. (a),(c) Two instances with four nodes, four edges and a two-colour alphabet; (b) an assignment satisfying all edges of the first, $\opt = 1$; (d) an assignment for the second leaving one edge unsatisfied, $\opt = 3/4$.}{fig:ch03:ulc}
\end{frame}

\begin{frame}\ft{Why the gap must be imperfect: label propagation}
\ona{Suppose an instance is \emph{fully} satisfiable. Then a satisfying labelling can be found in polynomial time: choose a vertex, try each of the $k$ labels for it, and propagate along a spanning tree, since each edge determines the label at its other end (Fig.~\ref{fig:ch03:prop}). For each starting label this takes time linear in the number of edges, and one checks whether the result satisfies all edges. Uniqueness thus makes perfect completeness easy, which is why the conjecture concerns instances of value $1-\zeta$ rather than $1$.}

\begin{frame}\ft{The Unique Games Conjecture}
\begin{conj}[Unique Games Conjecture, \citet{khot2002power}]\label{conj:ch03:ugc}
For every pair of constants $\zeta,\delta>0$ there is an alphabet size $k = k(\zeta,\delta)$ such that it is $\mathbf{NP}$-hard to decide whether a unique label cover instance with $k$ labels has optimum at least $1-\zeta$ or at most $\delta$.
\end{conj}
\ona{Three remarks help in reading it. The completeness $1-\zeta$ cannot be raised to $1$, by label propagation. The alphabet must be allowed to grow as $\zeta,\delta\to0$: with $k$ labels a random labelling already satisfies a $1/k$ fraction of the edges in expectation, so one needs $k > 1/\delta$. And with $k = 2$ the problem is the familiar one of deciding, for constraints $x_u = x_v$ or $x_u \ne x_v$, how many can be satisfied, which Goemans--Williamson approximates well; the conjecture concerns large alphabets.}
\onp{\begin{itemize}\item Completeness $1-\zeta$, not $1$ (propagation). \item $k > 1/\delta$ needed: random labels satisfy $1/k$. \item $k=2$ is MaxCut-like and approximable.\end{itemize}}
\end{frame}

\section{Equivalent and related forms of the conjecture}

\begin{frame}\ft{Reformulation 1: a game with two provers}
\ona{The name comes from \emph{two-prover one-round games}. A verifier holds a unique label cover instance. She picks an edge $e = (v,w)$ at random, sends $v$ to the first prover and $w$ to the second, receives labels $a$ and $b$, and accepts if $\pi_e(b) = a$. The provers may agree on a strategy beforehand but cannot communicate once the game begins. A deterministic strategy is a pair of labellings, so the largest acceptance probability, the \emph{value} of the game, equals $\opt(\mathcal L)$, and Conjecture~\ref{conj:ch03:ugc} says that the value of a unique game is hard to approximate.}
\ona{\begin{figure}[htb]}\begin{center}

\begin{frame}\ft{Reformulations 2 and 3: any graph, and the strong form}
\begin{itemize}\setlength\itemsep{0.35em}
    \item \textbf{Non-bipartite constraint graphs.} One may drop the bipartition: a unique game is a graph $G=(V,E)$, labels in $[k]$, and a bijection $\pi_{uv}$ on each edge with $\pi_{vu} = \pi_{uv}^{-1}$; an edge is satisfied if $x_v = \pi_{uv}(x_u)$. The conjecture for general graphs is equivalent to that for bipartite ones \citep{KhotSurvey}, and one may also assume the graph regular.
    \item \textbf{The strong form} \citep{khot2008vertex}. For non-bipartite instances, the ``yes'' case asks for a labelling and a set $X_0$ of at least $(1-\zeta)|X|$ \emph{vertices} such that all constraints between vertices of $X_0$ are satisfied. The ``no'' case asks that for every assignment of at most $t$ labels to each vertex, and every set of $\gamma|X|$ vertices, some constraint among them is violated. Khot and Regev showed that the original conjecture implies this apparently stronger one, and used it to show that vertex cover is hard to approximate within $2-\epsilon$.
\end{itemize}
\ona{Such relations are typical. Each formulation is shown to imply another by a reduction that preserves the gap, perhaps with a change of $\zeta$, $\delta$ and $k$, and one is then free to use whichever is most convenient.}
\end{frame}

\begin{frame}\ft{Reformulation 4: linear equations, or rotations on a clock}
\ona{The most concrete form is a system of linear equations. In $\Gamma$-Max-2Lin$(q)$ one is given equations $x_u - x_v = c_{uv} \pmod q$, each involving two variables, and asked to satisfy as many as possible. Each equation is a unique constraint, the bijection being a rotation by $c_{uv}$. \citet{khot2007optimal} showed that the Unique Games Conjecture is equivalent to its special case for such equations:}
\begin{center}
\emph{For all $\zeta,\delta>0$ there is $q$ such that it is $\mathbf{NP}$-hard to decide whether $1-\zeta$ or at most $\delta$ of a system of equations $x_u - x_v = c_{uv} \pmod q$ can be satisfied.}
\end{center}

\begin{frame}\ft{Interlude: expansion and expanders}
\ona{The last two reformulations are phrased in terms of how well sets of vertices in a graph are connected to the rest, so we pause to introduce the notions. Let $G = (V,E)$ be a $d$-regular graph on $n$ vertices; each vertex has $d$ edges. For a set $S \subseteq V$, write $E(S,\overline S)$ for the set of edges with one endpoint in $S$ and the other outside, and define the \emph{expansion} of $S$ as}
\begin{equation}\label{eq:ch03:phi}
    \Phi(S) \;=\; \frac{|E(S,\overline S)|}{d\,|S|} \;\in\; [0,1].
\end{equation}
\ona{The denominator counts all edge ends at vertices of $S$, so $\Phi(S)$ is the probability that a random edge leaving a random vertex of $S$ ends outside $S$. A set with $\Phi(S)$ near $0$ is almost closed off, like a community in a social network; a set with $\Phi(S)$ near $1$ sends almost all its edges out. The \emph{expansion of the graph} is that of its worst set of at most half the vertices,}
\begin{equation*}
    \Phi_G \;=\; \min_{0<|S|\le n/2}\Phi(S),
\end{equation*}
\ona{and a family of $d$-regular graphs, $d$ fixed and $n\to\infty$, is a family of \emph{expanders} if $\Phi_G \ge \varepsilon$ for some constant $\varepsilon>0$ and all members. Expanders are sparse, with $dn/2$ edges, and yet every set of at most half the vertices sends a constant fraction of its edges out.}

\begin{frame}\ft{Interlude, continued: examples, eigenvalues, and small sets}
\onp{\scriptsize}
\begin{itemize}\setlength\itemsep{0.3em}
    \item \textbf{Examples.} The cycle has $\Phi_G = 2/n$ and a $\sqrt n\times\sqrt n$ grid about $1/\sqrt n$: graphs with geometry are poor expanders, since a ball has a small boundary. The hypercube $\{0,1\}^m$, with $n = 2^m$ and $d = m$, has $\Phi_G = 1/m = 1/\log_2 n$, attained by a half-cube $\{x : x_1 = 0\}$. Random $d$-regular graphs with $d\ge3$ are expanders with high probability, and explicit constructions exist.
    \item \textbf{Eigenvalues.} Let $1 = \lambda_1 \ge \lambda_2 \ge \dots \ge \lambda_n$ be the eigenvalues of the normalised adjacency matrix $A/d$. \emph{Cheeger's inequality} states that
    \begin{equation*}
        \tfrac12(1-\lambda_2) \;\le\; \Phi_G \;\le\; \sqrt{2(1-\lambda_2)}.
    \end{equation*}
    The \emph{spectral gap} $1-\lambda_2$ can be computed in polynomial time, so whether a graph is an expander can be decided, up to the square root, efficiently; and the eigenvector of $\lambda_2$ yields a set $S$ with $\Phi(S) \le \sqrt{2(1-\lambda_2)}$. On expanders, random walks forget their starting point after $\mathcal O(\log n)$ steps.
    \item \textbf{Small sets.} The square-root loss in Cheeger's inequality is unavoidable for spectral methods, and they give no control on the \emph{size} of the set found. The \emph{small-set expansion} of $G$ at size $\mu$ is $\min\{\Phi(S) : |S| = \mu n\}$. A graph can expand well on large sets and badly on small ones, or the reverse: in the hypercube, a subcube fixing $t$ of the $m$ coordinates has $n/2^t$ vertices and $\Phi(S) = t/m$.
\end{itemize}
\ona{The connection with unique games is now visible. By \eqref{eq:ch03:expansion} below, a good labelling is a small set, of relative size $1/k$, in the label-extended graph, which does not expand. If the constraint graph $G$ itself is an expander, such sets are rare and can be found: unique games on expanding constraint graphs are easy \citep{arora2008unique}. The hard instances, if any, must have constraint graphs with many small non-expanding sets, and the small-set expansion hypothesis asserts that finding such sets is hard.}
\end{frame}

\begin{frame}\ft{Reformulation 5: sets in the label-extended graph}
\ona{Replace each vertex $u$ of $G$ by $k$ copies $(u,1),\dots,(u,k)$, one per label, and each edge $uv$ by the perfect matching $(u,a)\sim(v,\pi_{uv}(a))$. The result is the \emph{label-extended graph} $G_{\mathcal L}$, a $k$-fold cover of $G$. A labelling $x$ is the same thing as a set $S_x = \{(u,x_u)\}$ containing exactly one copy of each vertex, and the edge $uv$ is satisfied precisely when its lifted edge from $(u,x_u)$ stays inside $S_x$. If $G$ is $d$-regular, so is $G_{\mathcal L}$, the set $S_x$ has a $1/k$ fraction of its vertices, and}
\begin{equation}\label{eq:ch03:expansion}
    \Phi(S_x) \;=\; \frac{|E(S_x, \overline{S_x})|}{d\,|S_x|} \;=\; 1 - \mathrm{val}(x).
\end{equation}
\ona{In the language of \eqref{eq:ch03:phi}, a labelling of value $1-\zeta$ is a set of relative size $1/k$, of a special shape, with expansion only $\zeta$: few of its edges leave it.}

\ona{Until 2026 the conjecture was neither proved nor refuted. On the other side: \citet{arora2015subexponential} gave an algorithm running in time $\exp(n^{\mathrm{poly}(\zeta)})$, so that, if the conjecture holds, any reduction from an $\mathbf{NP}$-complete problem must enlarge instances by a polynomial whose degree grows as $\zeta \to 0$; and unique games whose constraint graphs are good expanders are easy \citep{arora2008unique}. \section{The Unique Games Theorem (2026)}\label{sec:ch03:ugt}

\begin{frame}\ft{A claimed proof of the conjecture}
\onp{\footnotesize}
\ona{On 23 September 2026 OpenAI released a preprint, \emph{The Unique Games Theorem} \citep{OAI:The-Unique-Games-Theorem-September-23-2026}, which claims to prove Conjecture~\ref{conj:ch03:ugc}. Its main result reads as follows.}
\begin{thm}[Unique Games Theorem, claimed in \citet{OAI:The-Unique-Games-Theorem-September-23-2026}]\label{thm:ch03:ugt}
For every fixed $\epsilon,\delta\in(0,1/2)$ there are an integer $s\ge1$ and a deterministic polynomial-time reduction from 3-SAT to explicit, unweighted unique games $G_\varphi$ over the alphabet $K = \mathbb F_2^s$ such that $\varphi$ satisfiable implies $\mathrm{val}(G_\varphi)\ge1-\epsilon$, and $\varphi$ unsatisfiable implies $\mathrm{val}(G_\varphi)\le\delta$. The constraint graph is simple and bipartite, and every constraint is a translation $x_v = x_u + c_e$ in $K$.
\end{thm}
\begin{itemize}\setlength\itemsep{0.25em}
    \item \textbf{The quantifiers are those of the conjecture:} the two errors are fixed first, and then the alphabet. The constraints are linear equations over $\mathbb F_2^s$, the form of Reformulation~4, which was known to be equivalent.
    \item \textbf{Status.} As of October 2026 the preprint has not been peer reviewed, and the community has not yet reached a verdict on it. Until it has, every statement in this \chap{} that assumes the conjecture remains conditional, and we keep the label ``UG-hard''.
    \item \textbf{What it does not settle.} The small-set expansion hypothesis, which implies the conjecture, is not addressed. Nor are the gaps that remain without the conjecture's help, such as $[0.9326, 0.997]$ for MaxCut on 3-regular graphs.
\end{itemize}
\onp{\scriptsize Unreviewed preprint (October 2026); results that assume the conjecture stay conditional until it is checked.}
\end{frame}

\begin{frame}\ft{How the claimed proof is organised}
\onp{\scriptsize}
\ona{The proof is long, 58 pages, and builds on most of the work described in this \chap{}. Its outline, as the preprint describes it:}
\begin{itemize}\setlength\itemsep{0.25em}
    \item \textbf{Starting point.} H\accent23 astad's gap for nearly satisfiable systems of parity equations on three variables \citep{Hstad2001}.
    \item \textbf{The obstacle.} Completeness near $1$ is the difficult part when each label allows only one reply. The 2-to-2 theorem \citep{Khot2018} uses a test on binary matrices, which are perturbed by adding rank-one matrices. The natural unique version of this test keeps its answer with probability only about $1/2$, which is the source of the completeness $1/2$ mentioned above.
    \item \textbf{The new ingredient: a ``latent alphabet'' gadget.} For any $p>0$, the preprint constructs a nonlinear map $C$ from a larger binary space $V \supseteq K$ to $K$ and a noise distribution on $V$. The map commutes with translations by $K$, and changes its value under the noise with probability at most $p$. At the same time, any family of linear observations of large rank still detects the noise. The construction iterates quadratic maps over a finite field. Testing equality of $C$ before and after a rank-one perturbation restores completeness near $1$.
    \item \textbf{Soundness.} A labelling that passes the test often has Fourier mass on low-rank characters. An inverse theorem for the matrix shortcode, derived from the Grassmann expansion theorem of \citet{Khot2018}, then turns this mass into correlated strategies for a two-prover game. In the ``no'' case, parallel repetition, in the form of \citet{dinur2014analytical}, shows that such strategies agree too rarely. The two bounds contradict each other.
    \item \textbf{Finishing.} Rounding weights and subdividing edges give the explicit unweighted, simple bipartite instances of Theorem~\ref{thm:ch03:ugt}.
\end{itemize}
\end{frame}

\begin{frame}\ft{What would follow}
\ona{If Theorem~\ref{thm:ch03:ugt} is correct, every hardness result proved under the conjecture becomes an unconditional $\mathbf{NP}$-hardness result. Section~8 of the preprint works out the details for, among others, the following:}
\begin{itemize}\setlength\itemsep{0.25em}
    \item \textbf{MaxCut:} approximating within any ratio in $(\alpha_{\rm GW},1)$ is $\mathbf{NP}$-hard, by Theorem~\ref{thm:ch03:kkmo} below. Goemans--Williamson would then be optimal among polynomial-time algorithms, classical or quantum, unless $\mathbf{NP}\subseteq\mathbf{BQP}$ in the quantum case.
    \item \textbf{Vertex cover:} hard within every factor in $(1,2)$ \citep{khot2008vertex}.
    \item \textbf{Every CSP:} Raghavendra's semidefinite thresholds become $\mathbf{NP}$-hard \citep{raghavendra2008optimal}; for Max-2-SAT this gives $\approx0.940$ \citep{austrin2007balanced,brakensiek2024tight}.
    \item \textbf{Ordering problems:} maximum acyclic subgraph is hard beyond the ratio $1/2$, and betweenness beyond $1/3$.
    \item \textbf{Cut problems:} multicut, non-uniform sparsest cut and correlation clustering have no constant-factor approximation \citep{khot2015unique}.
\end{itemize}
\ona{For this course the first item is the one that matters: no polynomial-time quantum algorithm, QAOA included, could beat $\alpha_{\rm GW}\approx0.878$ on every MaxCut instance, unless $\mathbf{NP}\subseteq\mathbf{BQP}$. Any quantum advantage for MaxCut would then have to come from particular families of instances, or from exceeding polynomial time, as \Chap~\chref{C.EQA} argued for chemistry.}
\onp{\footnotesize If correct: GW optimal for MaxCut unless $\mathbf{NP}\subseteq\mathbf{BQP}$; quantum advantage only on instance families.}
\end{frame}

\section{Boolean functions: dictators, long codes and noise stability}

\begin{frame}\ft{A vocabulary for the hardness proof}
\onp{\scriptsize}
\ona{The hardness proof of the next section uses a few notions from the analysis of Boolean functions, for which \citet{odonnell2014analysis} is the standard reference; we follow his notation. A \emph{Boolean function} is a map $f:\{-1,1\}^k\to\{-1,1\}$, which one may think of as a voting rule: $k$ voters each vote $\pm1$, and $f$ declares the outcome.}
\begin{itemize}\setlength\itemsep{0.25em}
    \item \textbf{Fourier expansion.} Every such $f$ is a multilinear polynomial, $f(x) = \sum_{S\subseteq[k]}\hat f(S)\prod_{i\in S}x_i$, and by Parseval $\sum_S\hat f(S)^2 = \mathbb E[f^2] = 1$. The weights $\hat f(S)^2$ thus form a probability distribution over sets of voters.
    \item \textbf{Dictators and the long code.} The \emph{dictator} $f(x) = x_i$ lets voter $i$ decide alone. The \emph{long code} encodes a label $i\in[k]$ by the dictator $x\mapsto x_i$, that is, by its table of $2^k$ values. This is a hugely redundant code, $k$ labels written with $2^k$ bits, but its redundancy is what allows a verifier to check, by reading only two or three bits, whether a table is close to a codeword.
    \item \textbf{Influence.} The influence of voter $i$ is the probability that flipping the $i$-th vote changes the outcome, $\mathrm{Inf}_i[f] = \mathrm{Pr}_x[f(x)\ne f(x^{\oplus i})] = \sum_{S\ni i}\hat f(S)^2$. A dictator has $\mathrm{Inf}_i = 1$ for its dictator and $0$ for everyone else; the majority of $k$ votes, $k$ odd, gives each voter influence of order $1/\sqrt k$. A function has \emph{no notable coordinates} if every influence is at most a small $\tau$.
    \item \textbf{Noise stability.} Call $y$ \emph{$\rho$-correlated} with $x$ if each $y_i$ equals $x_i$ with probability $(1+\rho)/2$ and $-x_i$ otherwise, independently, so that $\mathbb E[x_iy_i] = \rho$. The \emph{noise stability} is $\mathrm{Stab}_\rho[f] = \mathbb E[f(x)f(y)] = \sum_S\rho^{|S|}\hat f(S)^2$, and $\mathrm{Pr}[f(x)\ne f(y)] = \tfrac12 - \tfrac12\mathrm{Stab}_\rho[f]$. For a dictator $\mathrm{Stab}_\rho = \rho$; for the majority, $\mathrm{Stab}_\rho[\mathrm{Maj}_k] \to 1 - \tfrac2\pi\arccos\rho$ as $k\to\infty$ (Sheppard's formula), the probability that two $\rho$-correlated Gaussian vectors fall on the same side of a random hyperplane.
\end{itemize}
\end{frame}

\begin{frame}\ft{\emph{Majority is Stablest}, and dictator tests}
\begin{thm}[Majority is Stablest; \citet{mossel2010noise}, see \citet{odonnell2014analysis}]\label{thm:ch03:mis}
For every $\rho\in(-1,1)$ and $\epsilon>0$ there is $\tau>0$ such that every $f:\{-1,1\}^k\to[-1,1]$ with $\mathrm{Inf}_i[f]\le\tau$ for all $i$ satisfies: if $0\le\rho<1$ and $\mathbb E[f]=0$, then $\mathrm{Stab}_\rho[f] \le 1-\tfrac2\pi\arccos\rho + \epsilon$; and if $-1<\rho\le0$, then $\mathrm{Stab}_\rho[f] \ge 1-\tfrac2\pi\arccos\rho - \epsilon$.
\end{thm}
\ona{Among all voting rules in which no single voter matters much, the majority is, asymptotically, the one whose outcome is least changed by noise. For negative $\rho$, which is the case we need, the statement reads: a function with no notable coordinates disagrees on $\rho$-correlated inputs with probability at most $\arccos(\rho)/\pi + \epsilon/2$, whereas a dictator disagrees with probability $(1-\rho)/2$.}
\ona{This difference is exploited by a \emph{dictator test}, or more precisely a \emph{Dictator-vs.-No-Notables test} \citep{odonnell2014analysis}: a randomised procedure that reads a few values of a function $f$ and accepts or rejects, such that}
\begin{itemize}\setlength\itemsep{0.25em}
    \item \textbf{completeness} $c$: every dictator is accepted with probability at least $c$;
    \item \textbf{soundness} $s$: every function with no notable coordinates is accepted with probability at most $s + o(1)$.
\end{itemize}
\ona{If the test checks a constraint of some CSP on the values it reads, then the CSP is UG-hard to approximate beyond $s/c$: composing the test with a unique game, by writing each label in the long code and checking edges through the bijections, turns the gap between dictators and functions with no notable coordinates into the gap between good and bad labellings \citep{khot2007optimal,odonnell2014analysis}. Proving hardness of approximation under the conjecture thus reduces to designing a test, and analysing it with Fourier analysis.}
\onp{\scriptsize Dictators accepted with probability $\ge c$; functions with no notable coordinates with $\le s+o(1)$ $\Rightarrow$ UG-hard beyond $s/c$.}
\end{frame}

\section{Why Goemans--Williamson is optimal, given the conjecture}

\begin{frame}\ft{The curve behind $\alpha_{\rm GW}$}
\ona{Recall from \Chap~\chref{C.Motivation} that random-hyperplane rounding separates two unit vectors at angle $\theta$ with probability $\theta/\pi$, while the relaxation counts the edge as $(1-\cos\theta)/2$. Writing $\rho = \cos\theta$ for the correlation of the two vectors,}
\begin{equation}\label{eq:ch03:alpha}
    \alpha_{\rm GW} \;=\; \min_{-1\le\rho<1}\frac{\arccos(\rho)/\pi}{(1-\rho)/2} \;\approx\; 0.8786, \qquad \text{attained at } \rho^\ast \approx -0.689.

\begin{frame}\ft{The dictatorship test and \emph{Majority is Stablest}}
\ona{The hardness proof of \citet{khot2007optimal} turns this curve into a Dictator-vs.-No-Notables test in the sense of the previous section. For a fixed $\rho \in (-1,0)$, test a function $f:\{\pm1\}^k\to\{\pm1\}$ as follows:}
\begin{itemize}\setlength\itemsep{0.25em}
    \item pick $x \in \{\pm1\}^k$ uniformly, and $y$ by setting $y_i = x_i$ with probability $(1+\rho)/2$ and $y_i = -x_i$ otherwise, independently;
    \item accept if $f(x) \ne f(y)$. This is a MaxCut edge between the points $x$ and $y$ of the cube.
\end{itemize}
\ona{A dictator passes with probability $(1-\rho)/2$, which is the completeness. A function with no notable coordinates passes with probability $\tfrac12-\tfrac12\mathrm{Stab}_\rho[f] \le \arccos(\rho)/\pi + o(1)$ by Theorem~\ref{thm:ch03:mis}, which is the soundness; the majority comes close to it. So the test distinguishes functions that encode a label from functions that encode none, with completeness and soundness given by the two curves of Fig.~\ref{fig:ch03:noise}.}
\onp{\footnotesize Dictators pass with probability $(1-\rho)/2$; functions with small influences with at most $\arccos(\rho)/\pi + \epsilon$ \citep{mossel2010noise}.}
\end{frame}

\begin{frame}\ft{From unique games to MaxCut}
\ona{Composing the dictatorship test with a unique game, by placing a long code at each vertex and checking edges through the bijections, gives a gap-preserving reduction:}
\begin{thm}[\citet{khot2007optimal}, with \citet{mossel2010noise}]\label{thm:ch03:kkmo}
Assume the Unique Games Conjecture. For every $\rho\in(-1,0)$ and $\epsilon>0$, it is $\mathbf{NP}$-hard to decide whether a MaxCut instance has value at least $(1-\rho)/2 - \epsilon$ or at most $\arccos(\rho)/\pi + \epsilon$. Taking $\rho = \rho^\ast$, it is $\mathbf{NP}$-hard to approximate MaxCut within $\alpha_{\rm GW} + \epsilon$.
\end{thm}

\begin{frame}\ft{Every constraint satisfaction problem}
\onp{\footnotesize}
\ona{The argument for MaxCut is one case of a general principle. For any CSP there is a canonical semidefinite relaxation, of which the Goemans--Williamson programme is the instance for MaxCut, and a canonical way of rounding it.}
\begin{thm}[\citet{raghavendra2008optimal}, informal]
Assume the Unique Games Conjecture. For every CSP, the largest ratio between the integral optimum and the value of the canonical semidefinite relaxation, over all instances, is also the best ratio achievable in polynomial time; and a generic rounding algorithm achieves it.
\end{thm}
\begin{itemize}\setlength\itemsep{0.25em}
    \item Under the UGC, semidefinite programming is thus the last word on approximating CSPs in polynomial time. The approximation ratio of an algorithm and the hardness threshold are two faces of the same integrality gap.
    \item Other consequences include the hardness of approximating vertex cover within $2-\epsilon$ \citep{khot2008vertex}, and of approximating non-uniform sparsest cut within any constant \citep{khot2015unique}.
    \item For quantum algorithms the conjecture sets the bar. If it holds, as the 2026 preprint claims, a polynomial-time quantum algorithm beating $\alpha_{\rm GW}$ on every instance would solve $\mathbf{NP}$-hard problems, which is not expected.
\end{itemize}
\end{frame}

\begin{frame}\ft{Beyond positive weights: QUBO}
\ona{The ratio $\alpha_{\rm GW}$ concerns MaxCut with non-negative weights. With real weights, that is, for general QUBO, the natural ratio changes:}
\begin{propn}[\citet{charikar2004maximizing}]
Let it be $\mathbf{NP}$-hard to decide whether the maximum cut of a MaxCut instance with total weight $w$ is at least $k$ or less than $\alpha k$. Then for every $\epsilon > 0$ it is $\mathbf{NP}$-hard to distinguish QUBO instances with optimum at least $2k - w$ from those with optimum at most $2\alpha k - w$, a ratio of $\alpha + w(\alpha-1)/(2k-w)$; the corresponding hardness factor for MaxCut with real weights is $11/13 \approx 0.846$, and it is achieved by randomised rounding of a semidefinite relaxation.
\end{propn}
\ona{This is why the histograms of \Chap~\chref{C.WSRounded} mark $11/13$: on weighted instances that, and not $0.878$, is the yardstick.}

\section{Structural obstructions and average-case hardness}

\begin{frame}\ft{Even the best angles do not help at constant depth}
\begin{itemize}\setlength\itemsep{0.3em}
    \item \textbf{Symmetry.} QAOA started from the uniform superposition inherits the $\mathbb{Z}_2$ spin-flip symmetry of $H_C$ and $H_M$, so $\langle Z_i\rangle = 0$; on bipartite $D$-regular graphs, GW provably beats QAOA at every constant depth $p$ \citep{Bravyi2020}.
    \item \textbf{Locality.} A circuit of depth $p$ sees only neighbourhoods of radius $p$; on graphs of high girth QAOA is a local algorithm, and fails on MAX-$k$-CUT \citep{BravyiKlieschKoenigTang2022,BarakMarwaha2022}.
    \item \textbf{Overlap gap.} Sparse random CSPs have the overlap gap property: any two near-optimal solutions are either close together or far apart. Every circuit of depth $p < \varepsilon\log n$ then falls short of the optimum with high probability \citep{ChouLoveSandhuShi2021,ChenHuangMarwaha2023}. This is average-case hardness for classical local algorithms and low-depth quantum algorithms alike; whether deeper quantum algorithms escape it is not known, and if they did, that would be a convincing route to an advantage.
\end{itemize}
\ona{Recursive QAOA \citep{Bravyi2020} and warm starts \citep{egger2021warm} break the symmetry and the locality; they are the subject of \Chaps~\chref{C.WSRounded}--\chref{C.WSContinuous}.}
\end{frame}

\begin{frame}\ft{Positive results, for balance}
\begin{itemize}\setlength\itemsep{0.2em}
    \item MaxCut on 3-regular graphs: ratio at least $0.692$ at $p=1$ \citep{farhi2014quantum}, $0.7559$ at $p=2$, and, under assumptions, $0.7924$ at $p=3$.
    \item On $d$-regular graphs of large girth, QAOA numerically outperforms semidefinite relaxations from $p \ge 11$.
    \item Warm starts inherit classical guarantees \citep{egger2021warm}; recursive QAOA reduces the instance one variable at a time using measured correlations \citep{Bravyi2020}.
    \item The travelling salesman problem as a reminder of nuance: $\mathbf{APX}$-hard in general, but with a PTAS in the Euclidean case and millions of nodes solved heuristically; while some problems with fewer than 100 variables, such as low-autocorrelation binary sequences, remain hard.
\end{itemize}
\ona{For the rest of the course the yardsticks are $0.878$ for MaxCut and $11/13$ for QUBO and, more important in practice, the performance of Goemans--Williamson and of the best classical solvers on the family of instances at hand: Gurobi, and the semidefinite branch-and-bound solvers BiqMac \citep{rendl2010solving}, BiqCrunch \citep{krislock2017biqcrunch}, BiqBin \citep{gusmeroli2022biqbin} and MADAM \citep{hrga2021madam}.}
\end{frame}
