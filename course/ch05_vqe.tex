\chapter{The Variational Quantum Eigensolver}\label{C.VQE}
\ona{\sloppy}

\ona{This \chap{} introduces variational quantum algorithms through their first instance, the variational quantum eigensolver (VQE) of \citet{peruzzo2014variational}, with the theory of \citet{mcclean2016theory} and the review of \citet{cerezo2021variational}. After the variational loop, we follow one run of it in full detail, depth-one QAOA on the 4-cycle, from the circuit and its controlled gates to the measured copies and the parameter-shift rule. We then explain where good ans\"atze come from: VQE as a discretisation of adiabatic computation, with a bound on the error of the discretisation and a comparison, for small hydrogen molecules, of the discretised schedule with VQE started from it. The main subject of the \chap{} is then what it means to \emph{train} such an algorithm, and how hard that is. We follow \citet{bittel2021training}, \emph{Training variational quantum algorithms is NP-hard}, who turn the training into a family of well-posed computational problems, with a measure of error that separates what the circuit cannot represent from what the optimizer fails to find. They show that the training is $\mathbf{NP}$-hard even when the quantum part is trivial: for circuits without entanglement, on logarithmically many qubits, with a single parameter, for QAOA, and for free fermions; and that in the worst case no polynomial-time algorithm can guarantee any non-trivial error. We explain the formalisation first, and then prove the results, all but one in full. Parts are adapted from our course on quantum computing \citep{aspman2024quantum}.}

\section{The variational principle}

\begin{frame}\ft{What we need from quantum mechanics}
\begin{itemize}\setlength\itemsep{0.3em}
    \item A \emph{state} of $n$ qubits is a unit vector $\ket\psi \in \C^{2^n}$, written in the computational basis $\{\ket{x} : x \in \{0,1\}^n\}$ as $\ket\psi = \sum_x \psi_x \ket x$; measuring returns $x$ with probability $|\psi_x|^2$.
    \item An \emph{observable} is a Hermitian matrix $O$; its expectation in $\ket\psi$ is $\braket{\psi|O|\psi}$. Every $O$ on $n$ qubits is a real combination of Pauli strings $P \in \{I,X,Y,Z\}^{\otimes n}$, and $\braket{P}$ is estimated by measuring each qubit in the corresponding basis and averaging $\pm1$ outcomes.
    \item Evolution is by unitaries; a gate $e^{-i\theta H}$ with $H$ Hermitian is a parametrised unitary.
\end{itemize}
\begin{thm}[Variational principle]\label{thm:ch05:variational}
Let $H \in \Herm(\C^d)$ have smallest eigenvalue $E_0$. For every non-zero $\ket\psi$, $\braket{\psi|H|\psi}/\braket{\psi|\psi} \ge E_0$, with equality iff $\ket\psi$ is a ground state.
\end{thm}
\ona{\emph{Proof.} Expand $\ket\psi = \sum_k C_k\ket{\phi_k}$ in an eigenbasis, $H\ket{\phi_k} = E_k\ket{\phi_k}$. Then $\braket{\psi|H|\psi} = \sum_k|C_k|^2E_k \ge E_0\sum_k|C_k|^2 = E_0\braket{\psi|\psi}$, with equality iff $C_k = 0$ whenever $E_k > E_0$. \qed}
\end{frame}

\section{Variational quantum algorithms}

\begin{frame}\ft{The variational loop}
\onp{\footnotesize}
\onp{\begin{columns}[T]\begin{column}{0.6\textwidth}}
\ona{A \emph{variational quantum algorithm} (VQA) restricts $\ket\psi$ to states prepared by a parametrised circuit, $\ket{\Psi(\vartheta)} = U(\vartheta)\ket{\Psi_0}$, and minimises the upper bound $C(\vartheta) = \braket{\Psi(\vartheta)|H|\Psi(\vartheta)}$ with a classical optimizer, Fig.~\ref{fig:ch05:vqa}. Concretely, a VQA is specified by:}
\begin{itemize}\setlength\itemsep{0.15em}
    \item a dataset $\mathcal D$ and its quantum embedding, here the Hamiltonian $H = \sum_k h_k P_k$ in the Pauli basis;
    \item a parametrised circuit $U(\vartheta) = U_L(\vartheta_L)\cdots U_1(\vartheta_1)$, $U_i = e^{-i\vartheta_iH_i}$, and an initial state $\ket{\Psi_0}$: the \emph{ansatz};
    \item a cost $C(\vartheta) = \sum_k h_k\braket{\Psi(\vartheta)|P_k|\Psi(\vartheta)}$, estimated from repeated measurements up to an additive error $\epsilon \sim 1/\sqrt{\text{shots}}$;
    \item a classical optimizer $\mathcal A$ for $\min_\vartheta C(\vartheta)$: derivative-free (COBYLA, SPSA) or gradient-based via the parameter-shift rule (\Chap~\chref{C.PSR}).
\end{itemize}

\includegraphics[height=0.74\textheight,width=\linewidth,keepaspectratio]{figures/vqa/vqa2024.pdf}
\end{column}\end{columns}}

\section{A worked example: depth-one QAOA on the 4-cycle}\label{sec:ch05:worked}

\begin{frame}\ft{A worked example: the instance and the ansatz}
\onp{\scriptsize}
\onp{\begin{columns}[T]\begin{column}{0.66\textwidth}}
\ona{To see every step of the loop of Fig.~\ref{fig:ch05:vqa} at once, we run a few iterations on the smallest interesting case: MaxCut on the 4-cycle, with vertices $1,2,3,4$ and edges $12$, $23$, $34$, $41$, and the depth-one ansatz of the quantum approximate optimization algorithm, which \Chap~\chref{C.QAOA} studies in general. With the conventions of that \chap{},}
\begin{gather}
    H_C = -\tfrac12\sum_{ij\in E}(\one - Z_iZ_j),\qquad H_M = \sum_{j=1}^4 X_j,\nonumber\\
    \ket{\psi(\beta,\gamma)} = e^{-i\beta H_M}e^{-i\gamma H_C}\ket{+}^{\otimes4}.
\end{gather}
\ona{The cost $F(\beta,\gamma) = \braket{\psi(\beta,\gamma)|H_C|\psi(\beta,\gamma)}$ is minus the expected cut of a measured bit string. A short computation (Exercise~\ref{exe:ch05:c4}) gives}
\begin{equation}\label{eq:ch05:c4}
    F(\beta,\gamma) = -2 + \sin4\beta\,\sin2\gamma,
\end{equation}
\ona{with minimum $-3$ at $(\beta,\gamma) = (\pi/8,-\pi/4)$: at depth one, the expected cut is at most $3$ of the $4$ edges. The optimizer does not know \eqref{eq:ch05:c4}; it sees only estimates from measurements. Every number below is produced by \texttt{code/ch05\_worked\_example.py}, which simulates the circuit gate by gate and samples its measurements.}
\begin{itemize}\setlength\itemsep{0.1em}
    \item two parameters $(\beta,\gamma)$, starting at $(0.1,-0.1)$;
    \item $M = 1000$ measured copies for every circuit;
    \item gradients by the parameter-shift rule; steps $(\beta,\gamma)\leftarrow(\beta,\gamma) - 0.1\,\hat g$.
\end{itemize}

\begin{frame}\ft{The circuit, gate by gate}
\onp{\footnotesize}
\ona{The terms of $H_C$ commute, so up to a global phase $e^{-i\gamma H_C} = \prod_{ij\in E}R_{ZZ}(\gamma)$, with $R_{ZZ}(\gamma) = e^{-i\gamma Z_iZ_j/2}$ on qubits $i$ and $j$. A $ZZ$ rotation is not native on most hardware, and is built from controlled gates. Since conjugation by a CNOT maps $\one\otimes Z$ to $Z\otimes Z$,}
\begin{equation}\label{eq:ch05:rzz}
    R_{ZZ}(\gamma) = \mathrm{CNOT}\,\big(\one\otimes R_Z(\gamma)\big)\,\mathrm{CNOT} = e^{i\gamma/2}\big(R_Z(\gamma)\otimes R_Z(\gamma)\big)\,\mathrm{CP}(-2\gamma),
\end{equation}
\ona{with $R_Z(\gamma) = e^{-i\gamma Z/2}$ and the controlled phase rotation $\mathrm{CP}(\varphi) = \mathrm{diag}(1,1,1,e^{i\varphi})$: either two CNOTs around a $Z$ rotation of the target, or one controlled rotation and two single-qubit ones (Exercise~\ref{exe:ch05:rzz}). The mixer is $e^{-i\beta H_M} = \prod_jR_X(2\beta)$, with $R_X(\theta) = e^{-i\theta X/2}$. In all, Fig.~\ref{fig:ch05:circuit}: four Hadamard gates, eight CNOTs, four $R_Z(\gamma)$ and four $R_X(2\beta)$, then a measurement of every qubit.}

\begin{frame}\ft{One evaluation needs many copies}
\onp{\footnotesize}
\ona{A measurement of all four qubits returns one bit string $x$, with probability $|\braket{x|\psi(\beta,\gamma)}|^2$, and destroys the state. The energy is therefore estimated from $M$ copies, each prepared from scratch and measured once:}
\begin{equation}
    \hat F = -\frac1M\sum_{m=1}^M\mathrm{cut}(x_m), \qquad\text{with standard error}\quad \frac{\hat\sigma}{\sqrt M}.
\end{equation}
\ona{At the starting point $(0.1,-0.1)$, the first twelve of the $M = 1000$ copies gave the bit strings $x_1x_2x_3x_4$ below, with their cuts. Of all $1000$, $98$ cut no edge, $765$ two and $137$ all four, against the probabilities $0.106$, $0.749$ and $0.145$ (Fig.~\ref{fig:ch05:shots}). The estimate $\hat F = -(2\cdot765 + 4\cdot137)/1000 = -2.078\pm0.031$ is within one standard error of the exact value $-2.077$. Halving the error costs four times as many copies; \Chap~\chref{C.PerIteration} counts them.}

\begin{frame}\ft{The gradient, by the parameter-shift rule}
\onp{\footnotesize}
\ona{The energy depends on the angle $\theta$ of a single gate $R_P(\theta) = e^{-i\theta P/2}$, with $P$ a Pauli string, as $a + b\cos\theta + c\sin\theta$; hence, exactly, $\partial F/\partial\theta = \frac12[F(\theta+\frac\pi2) - F(\theta-\frac\pi2)]$, the parameter-shift rule of \Chap~\chref{C.PSR}. Here $\gamma$ is the angle of four gates $R_Z(\gamma)$, one per edge, and $2\beta$ that of four gates $R_X(2\beta)$, one per qubit. Shifting one gate at a time, by the chain rule,}
\onp{For one gate $R_P(\theta) = e^{-i\theta P/2}$: $\partial F/\partial\theta = \frac12[F(\theta+\frac\pi2) - F(\theta-\frac\pi2)]$ (\Chap~\chref{C.PSR}). Here $\gamma$ sits in four $R_Z(\gamma)$ and $2\beta$ in four $R_X(2\beta)$; shift one gate at a time:}
\begin{equation}
    \frac{\partial F}{\partial\gamma} = \sum_{e\in E}\tfrac12\big[F_{e+} - F_{e-}\big], \qquad \frac{\partial F}{\partial\beta} = \sum_{j=1}^4 2\cdot\tfrac12\big[F_{j+} - F_{j-}\big],
\end{equation}
\ona{where $F_{e\pm}$ is the energy with only the $R_Z$ of edge $e$ shifted by $\pm\pi/2$, and $F_{j\pm}$ the energy with only the $R_X$ of qubit $j$ shifted by $\pm\pi/2$. That is $16$ circuits, each estimated from $1000$ copies, Table~\ref{tab:ch05:psr}. By symmetry, the exact contributions of all edges are equal, $0.191$, as are those of all qubits, $-0.183$; the estimates scatter around them by shot noise.}

\begin{frame}\ft{What the worked example shows}
\onp{\footnotesize}
\begin{itemize}\setlength\itemsep{0.25em}
    \item \textbf{The cost is in the copies.} Six steps used $6\cdot17 = 102$ circuits and $102\,000$ copies. With the hardware figures of \Chap~\chref{C.PerIteration}, about $100\,\mu$s per copy and $6$~s of latency per submitted circuit, that is $10$~s of sampling and $10$ minutes of latency.
    \item \textbf{Noise sets the endgame.} The standard error of an energy is $0.03$, that of a gradient component $0.04$ to $0.09$; near the optimum, the gradient drowns in it.
    \item \textbf{The answer is a bit string.} At the last iterate a maximum cut is measured with probability $0.51$: $526$ of $1000$ copies. The output of the algorithm is the best bit string seen, not the energy.
    \item \textbf{Depth one caps the result.} No angles give an expected cut above $3$ of $4$; more layers can do better (\Chap~\chref{C.QAOA}).
    \item \textbf{The landscape here is benign}: two parameters, a single minimum in the region of interest, and a start near a saddle. The theorems of \citet{bittel2021training}, later in this \chap{}, show how different the worst case is.
\end{itemize}
\ona{And the instance is, of course, classically trivial: the 4-cycle is bipartite. The point of the example is the anatomy of one run, which is the same at a hundred qubits.}
\end{frame}

\begin{frame}\ft{VQE for molecules}
\ona{The variational quantum eigensolver \citep{peruzzo2014variational} is the VQA for the electronic-structure problem of \Chap~\chref{C.Motivation}: $H$ is the second-quantised molecular Hamiltonian mapped to qubits, with $\mathcal O(N^4)$ Pauli terms for $N$ orbitals, and the goal is the ground-state energy. Design choices:}
\begin{description}\setlength\itemsep{0.2em}
    \item[Initial state] the Hartree--Fock determinant $\ket{\Psi_0}$, a computational basis state, whose energy is already the mean-field energy; the variational circuit adds correlation.
    \item[Chemically inspired ansatz] unitary coupled cluster with singles and doubles (UCCSD), $U(\vartheta) = e^{T(\vartheta) - T(\vartheta)^\dagger}$ with excitation operators $T$; exact in a limit, deep in practice.
    \item[Hardware-efficient ansatz] layers of single-qubit rotations and native entangling gates \citep{kandala2017hardware}; shallow and expressive, but problem-agnostic.
    \item[Measurement] group commuting Pauli terms; the shot count to reach precision $\epsilon$ scales as $(\sum_k|h_k|)^2/\epsilon^2$ in the worst case.
\end{description}
\ona{On hardware, VQE has produced ground-state energies of small molecules (H$_2$, LiH, BeH$_2$) to chemical accuracy after error mitigation \citep{kandala2017hardware}. Its scaling to larger molecules is the subject of \Chap~\chref{C.EQA}.}
\end{frame}

\begin{frame}\ft{VQE as state preparation for early fault-tolerant algorithms}
\ona{VQE has no guarantee on the quality of its output state, and there is evidence that prepare-and-measure estimation as in VQE is not competitive for industrially relevant molecules: the shot counts to reach chemical accuracy are prohibitive. \citet{zhang2022computing} propose a division of labour that keeps what VQE is good at:}
\begin{enumerate}\setlength\itemsep{0.2em}
    \item run VQE (or Hartree--Fock, or a short adiabatic evolution) to prepare $\ket{\phi_0}$ with overlap $\eta = |\braket{\phi_0|\psi_0}|^2$;
    \item hand $\ket{\phi_0}$ to a ground-state property estimation routine that uses Hadamard tests of $e^{-ijH}$ and $O$ with classical Fourier post-processing, and returns $\braket{\psi_0|O|\psi_0}$ to accuracy $\epsilon$ with maximal evolution time $\widetilde{\mathcal O}(\gamma^{-1})$, regardless of how good $\ket{\phi_0}$ is, at total cost $\widetilde{\mathcal O}(\gamma^{-1}\epsilon^{-2}\eta^{-2})$ (\Chap~\chref{C.Motivation}).
\end{enumerate}
\ona{The variational state need not be accurate; it needs overlap. This reframes the goal of VQE from ``find the ground state'' to ``find a state with non-negligible $\eta$'', which is a lower bar, and connects to the warm starts of \Chaps~\chref{C.WSRounded}--\chref{C.WSContinuous}, where the initial state likewise needs only a good overlap with the optimum. Example properties: entries $D_{pq} = \braket{\psi_0|a_p^\dagger a_q|\psi_0}$ of the one-particle reduced density matrix, hence charge densities and dipole moments, each a combination of four Pauli-string expectations.}
\onp{VQE supplies overlap $\eta$; a Fourier-based post-processing of Hadamard tests supplies the guarantee \citep{zhang2022computing}. The variational state need not be accurate, only overlapping.}
\end{frame}

\section{VQE as discretised adiabatic computation}\label{sec:ch05:adiabatic}

\begin{frame}\ft{From an adiabatic path to an ansatz}
\onp{\footnotesize}
\ona{Where do good ans\"atze come from? The answer that comes with a guarantee is adiabatic computation (\Chap~\chref{C.Annealing}). Let $H_0$ be a Hamiltonian whose ground state $\ket{\Psi_0}$ is easy to prepare, let $H_1$ be the Hamiltonian whose ground state we want, and evolve slowly along the path}
\begin{equation}\label{eq:ch05:path}
    H(s) = (1-s)\,H_0 + s\,H_1,\qquad i\frac{d}{dt}\ket{\psi(t)} = H(t/T)\ket{\psi(t)},\quad \ket{\psi(0)} = \ket{\Psi_0},\quad 0\le t\le T.
\end{equation}
\ona{By the adiabatic theorem of \Chap~\chref{C.Annealing}, a total time $T$ of the order of $1/\Delta_{\min}^2$, where $\Delta_{\min}$ is the smallest gap along the path, leaves $\ket{\psi(T)}$ close to the ground state of $H_1$; write $\varepsilon_{\rm ad}(T) = 1 - |\braket{g|\psi(T)}|^2$ for the remaining infidelity, with $\ket g$ the ground state of $H_1$. For the electronic structure problem, $H_1 = H$ is the molecular Hamiltonian and the natural $H_0$ is the \emph{Fock operator} $F = \sum_p\varepsilon_p\,a_p^\dagger a_p$ of the Hartree--Fock method, with the orbital energies $\varepsilon_p$; among states with the right numbers of electrons, its ground state is the Hartree--Fock determinant $\ket{\rm HF}$, the usual initial state of VQE. A gate-based computer cannot apply \eqref{eq:ch05:path} directly. Three steps turn the evolution into a circuit, and then into VQE:}
\begin{enumerate}\setlength\itemsep{0.15em}
    \item \textbf{Piecewise constant:} on the $k$-th of $L$ intervals of length $\Delta t = T/L$, replace $H(t/T)$ by $H(s_k)$, with $s_k = (k-\frac12)/L$.
    \item \textbf{Trotter splitting:} replace $e^{-i\Delta t H(s_k)}$ by $e^{-i\beta_kH_0}\,e^{-i\gamma_kH_1}$, with $\gamma_k = s_k\Delta t$ and $\beta_k = (1-s_k)\Delta t$.
    \item \textbf{Free the angles:} keep the circuit, and let an optimizer choose all $2L$ angles $(\boldsymbol\beta,\boldsymbol\gamma)$:
\end{enumerate}
\begin{equation}\label{eq:ch05:hva}
    \ket{\Psi(\boldsymbol\beta,\boldsymbol\gamma)} = e^{-i\beta_LH_0}e^{-i\gamma_LH_1}\cdots e^{-i\beta_1H_0}e^{-i\gamma_1H_1}\ket{\Psi_0}.
\end{equation}
\ona{The first factor to act is $e^{-i\gamma_1H_1}$, since $e^{-i\beta_1H_0}$ would only multiply $\ket{\Psi_0}$ by a phase. Steps 1 and 2 cost an error that we bound next; step 3 can only help, as the schedule is one point in the parameter space.}
\end{frame}

\begin{frame}\ft{Steps 1 and 2: what the discretisation costs}
\onp{\footnotesize}
\begin{lem}\label{lem:ch05:duhamel}
If $H(t)$ and $\tilde H(t)$ generate the unitaries $U(t)$ and $\tilde U(t)$ on $[0,T]$, then $\|U(T) - \tilde U(T)\|\le\int_0^T\|H(t) - \tilde H(t)\|\,dt$.
\end{lem}
\ona{\emph{Proof.} From $U' = -iHU$ and $\tilde U' = -i\tilde H\tilde U$, $\frac d{dt}(U^\dagger\tilde U) = iU^\dagger(H - \tilde H)\tilde U$. Integrate from $0$ to $T$ and use that unitaries preserve norms. \qed}
\begin{lem}[First-order Trotter error]\label{lem:ch05:trotter}
For Hermitian $A$ and $B$, $\|e^{-i(A+B)} - e^{-iA}e^{-iB}\|\le\frac12\|[A,B]\|$.
\end{lem}
\ona{\emph{Proof.} Let $V(t) = e^{iBt}e^{iAt}e^{-i(A+B)t}$, so that the left-hand side is $\|V(1) - \one\|$. Then $V'(t) = ie^{iBt}e^{iAt}\big(e^{-iAt}Be^{iAt} - B\big)e^{-i(A+B)t}$, and $\|e^{-iAt}Be^{iAt} - B\|\le t\|[A,B]\|$, since the derivative of $e^{-iA\tau}Be^{iA\tau}$ is $-ie^{-iA\tau}[A,B]e^{iA\tau}$. Integrating, $\|V(1) - \one\|\le\int_0^1t\,\|[A,B]\|\,dt = \frac12\|[A,B]\|$. \qed}
\ona{Apply both to the path \eqref{eq:ch05:path}. In step 1, $\|H(t/T) - H(s_k)\| = |t/T - s_k|\,\|H_1 - H_0\|\le\|H_1-H_0\|/(2L)$ on the $k$-th interval, so by Lemma~\ref{lem:ch05:duhamel} the error is at most $T\|H_1-H_0\|/(2L)$. In step 2, Lemma~\ref{lem:ch05:trotter} with $A = \gamma_kH_1$ and $B = \beta_kH_0$ bounds the error of layer $k$ by $\frac12\beta_k\gamma_k\|[H_0,H_1]\|\le\frac18\Delta t^2\|[H_0,H_1]\|$, as $s_k(1-s_k)\le\frac14$; errors of products of unitaries add, so the $L$ layers cost at most $T^2\|[H_0,H_1]\|/(8L)$. In all,}
\begin{equation}\label{eq:ch05:disc}
    \big\|\ket{\psi(T)} - \ket{\Psi(\boldsymbol\beta^{\rm ad},\boldsymbol\gamma^{\rm ad})}\big\| \;\le\; \eta_L \coloneqq \frac{T\,\|H_1 - H_0\|}{2L} + \frac{T^2\,\|[H_0,H_1]\|}{8L},
\end{equation}
\ona{where $(\boldsymbol\beta^{\rm ad},\boldsymbol\gamma^{\rm ad})$ are the angles of the schedule. Both terms fall like $1/L$; second-order splittings would make the second fall like $1/L^2$.}
\onp{\footnotesize Steps 1 and 2 cost $\eta_L$: both terms fall like $1/L$; commuting $H_0, H_1$ would make step 2 exact.}
\end{frame}

\begin{frame}\ft{Step 3: the ansatz contains the adiabatic algorithm}
\onp{\scriptsize}
\begin{propn}\label{propn:ch05:adiabatic}
At the angles of the schedule, the state \eqref{eq:ch05:hva} has approximation error, relative to the spectral width of $H_1$, at most
\begin{equation*}
    \delta \;\le\; \varepsilon_{\rm ad}(T) + \eta_L.
\end{equation*}
Hence the model mismatch $\delta_m$ of the ansatz with $L$ layers obeys the same bound, and so does the result of any optimizer started at these angles that never increases the energy.
\end{propn}
\ona{\emph{Proof.} For any state $\ket\phi$ with $|\braket{g|\phi}|^2 = 1-\epsilon$, expanding in eigenvectors gives $\braket{\phi|H_1|\phi} - E_0\le\epsilon\,w(H_1)$, so $\ket{\psi(T)}$ has $\delta\le\varepsilon_{\rm ad}(T)$. For unit vectors $\phi,\phi'$ and any $c\in\R$, $|\braket{\phi|H_1|\phi} - \braket{\phi'|H_1|\phi'}|\le2\|H_1 - c\one\|\,\|\phi-\phi'\|$; with $c$ the midpoint of the spectrum, $2\|H_1 - c\one\| = w(H_1)$, and \eqref{eq:ch05:disc} adds at most $\eta_L$ to $\delta$. \qed}
\begin{itemize}\setlength\itemsep{0.2em}
    \item \textbf{A polynomial ansatz with a guarantee.} If the gap along the path is at least inverse-polynomial and the norms are polynomial, then $T$ and $L = \mathcal O\big((T\|H_1-H_0\| + T^2\|[H_0,H_1]\|)/\epsilon\big)$ are polynomial, and the ansatz contains a state with $\delta\le\epsilon$. As adiabatic computation is universal \citep{aharonov2007adiabatic}, this family of ans\"atze is too.
    \item \textbf{A certified starting point.} The schedule gives angles with a known error. An optimizer started there can only improve on them: a warm start in the sense of \Chaps~\chref{C.WSRounded}--\chref{C.WSContinuous}, whose guarantee survives the persistent local minima of \citet{bittel2021training}, although it does not promise the global optimum.
    \item \textbf{QAOA is the special case} $H_0 = -H_M$, $H_1 = H_C$, $\ket{\Psi_0} = \ket+^{\otimes n}$ \citep{farhi2014quantum}, up to the sign of $\boldsymbol\beta$; the worked example above is one layer of it.
\end{itemize}
\end{frame}

\begin{frame}\ft{The gates, for a molecule}
\onp{\footnotesize}
\ona{Under the Jordan--Wigner transformation (\Chap~\chref{C.Motivation}), $a_p^\dagger a_p = \frac12(\one - Z_p)$, so the two kinds of layer in \eqref{eq:ch05:hva} are very different.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item \textbf{The Fock layer is cheap.} $F = \mathrm{const} - \frac12\sum_p\varepsilon_pZ_p$, so $e^{-i\beta F}$ is, up to a phase, a product of single-qubit rotations $R_Z(-\beta\varepsilon_p)$.
    \item \textbf{The molecular layer is not.} $H = \sum_\alpha h_\alpha P_\alpha$ has $\mathcal O(N^4)$ Pauli strings for $N$ orbitals, which do not commute. One more Trotter splitting, $e^{-i\gamma H}\approx\prod_\alpha e^{-i\gamma h_\alpha P_\alpha}$, turns each factor into a Pauli rotation: a ladder of CNOTs around one $R_Z$, as in \eqref{eq:ch05:rzz}.
    \item \textbf{Untying the angles.} Giving each group of commuting terms its own angle in each layer yields the \emph{Hamiltonian variational ansatz} of \citet{Wecker2015}: more parameters, a richer ansatz, and a harder optimization.
    \item \textbf{H$_2$ is too easy.} In the minimal basis, the states reachable from $\ket{\rm HF}$ span two dimensions. One layer with free angles, $e^{-i\beta F}e^{-i\gamma H}\ket{\rm HF}$, already reaches the exact ground energy, while the schedule itself, with $T = 16$, needs $L = 9$ layers for chemical accuracy, $1.6$ mHa.
\end{itemize}
\ona{All numbers in this section come from \texttt{code/ch05\_adiabatic.py}, which builds the qubit Hamiltonians from the integrals computed by PySCF, as in \Chap~\chref{C.Motivation}, checks their ground energies against the full configuration interaction of PySCF, and works in the sector with the right numbers of spin-up and spin-down electrons, which both $F$ and $H$ preserve.}
\end{frame}

\begin{frame}\ft{What the adiabatic view gives, and what it does not}
\onp{\footnotesize}
\ona{The adiabatic view is the clearest justification of VQE and QAOA that we have, and it is worth being precise about its scope.}
\begin{itemize}\setlength\itemsep{0.25em}
    \item \textbf{It gives} an ansatz with a bound on its model mismatch, in terms of the gap, the norms and the commutator (\ona{Proposition~\ref{propn:ch05:adiabatic}}\onp{the proposition}); a starting point with a certified error; and universality.
    \item \textbf{It does not give the gap.} For molecules the gap along the path from the Fock operator is often large, but it shrinks as bonds are stretched\ona{, as in Fig.~\ref{fig:ch05:adiabaticT}}; for hard optimization problems it is exponentially small (\Chap~\chref{C.Annealing}), and so are the guarantees.
    \item \textbf{It does not make training easy.} Freeing the angles is what lets VQE beat the schedule with few layers, but it also exposes the optimizer to the landscapes of \citet{bittel2021training}, below; \ona{Fig.~\ref{fig:ch05:adiabaticL}}\onp{the H$_4$ example} already shows a local minimum.
    \item \textbf{It does not count the measurements.} The bounds are on states; each energy is still estimated from copies, as in the worked example, with the shot counts of \Chap~\chref{C.PerIteration}.
\end{itemize}
\ona{The same picture underlies the convergence proof for QAOA as the depth grows (\Chap~\chref{C.QAOA}), and the warm starts of \Chaps~\chref{C.WSRounded}--\chref{C.WSContinuous}, which replace the adiabatic starting point by one that comes from a classical algorithm.}
\end{frame}

\section{What kind of algorithm is a VQA?}

\begin{frame}\ft{VQAs as local search algorithms}
\ona{\textbf{VQAs as local search algorithms}.}
VQAs may be argued to be local search algorithms. Recall how local search works. Having selected the stop criteria, local search generates an initial random solution $s^* = s_0$ and evaluates it, $f(s_0)=f(s^*)$. While the stop criteria are not achieved, local search perturbs $s_0$ by $\delta$, with $\delta$ small. If $f(s_{\rm upd.}) < f(s_{0})$ store $s^* = s_{\rm upd.}$. This iterates until the stop criteria are satisfied and returns the lastly stored $s^*$.

Similarly, for VQAs after selecting stop criteria (usually number of iterations) the VQA begins with a randomized variational parameter $\vartheta^* = \vartheta_0$ and computes $C(\vartheta_0)$. While the stopping criteria are not satisfied it updates to $\vartheta_{\rm upd.}$ by a step of the classical optimizer. If $C(\vartheta_{\rm upd.}) < C(\vartheta^*)$ it stores $\vartheta^* = \vartheta_{\rm upd.}$ and iterates until the stopping criteria are satisfied and returns the lastly stored $\vartheta^*$.
\end{frame}

\begin{frame}\ft{VQAs are not approximation algorithms}
\ona{\textbf{VQAs are not approximation algorithms}.}
However, one may argue that VQAs cannot be viewed as \emph{approximation algorithms}. The reason is that an approximation algorithm for a problem $P$ is a polynomial-time algorithm that computes a solution within a guaranteed factor of the optimum for \emph{every} instance of the problem. For example, the Goemans--Williamson algorithm is an approximation algorithm which for Max-Cut provides an instance-independent performance ratio of $0.878$; such a result does not exist for VQAs in general.

\ona{There are exceptions: for MaxCut on 3-regular graphs, depth-one QAOA achieves ratio at least $0.692$ \citep{farhi2014quantum}, and warm-started QAOA inherits the ratio of the relaxation it starts from \citep{egger2021warm}; see \Chaps~\chref{C.QAOA} and~\chref{C.WSRounded}.}
\ona{To turn such statements into theorems, one has to say precisely which computational problem the classical optimizer is asked to solve, and how its failure is measured. This is what \citet{bittel2021training} do, and the rest of this \chap{} follows them.}
\end{frame}

\section{Training as a computational problem}\label{sec:ch05:formal}

\begin{frame}\ft{What a formalisation must fix}
\onp{\footnotesize}
\ona{To prove that training is hard, one must first say what training is. The loop of Fig.~\ref{fig:ch05:vqa} leaves much unspecified, and \citet{bittel2021training} fix it as follows (Fig.~\ref{fig:ch05:sketch}). From now on we write $\boldsymbol\phi$ for the parameters, as they do.}
\begin{itemize}\setlength\itemsep{0.1em}
    \item \textbf{The ansatz:} a product $U_L(\phi_L)\cdots U_1(\phi_1)$ of \emph{layers} $U_i(\phi_i) = e^{-i\phi_iH_i}$, each generated by a given Hermitian $H_i$ and controlled by one real parameter, $\boldsymbol\phi\in\R^L$.
    \item \textbf{The input:} how $H_i$ and $O$ are written down, which fixes the size of an instance.
    \item \textbf{The quantum part:} an oracle that returns $\braket{O(\boldsymbol\phi)}$, or a classical computation where that is efficient.
    \item \textbf{A solution:} parameters whose energy is optimal up to an exponentially small error.
    \item \textbf{The error:} relative to the spectral width of $O$, and split into the part due to the ansatz and the part due to the optimizer.
\end{itemize}
\figslide[height=0.3\textheight]{figures/bittel/sketch.pdf}{A variational routine, after \citet{bittel2021training}: parametrised layers $U_i(\phi_i)$ act on $\ket0^{\otimes N}$, a fixed unitary $U_{\rm M}$ and measurements in the computational basis estimate $\braket{O(\boldsymbol\phi)}$ after post-processing, and a classical optimizer proposes new parameters $\boldsymbol\phi_{\rm guess}$ until it returns $\boldsymbol\phi_{\rm final}$. The formalisation isolates the classical optimization (red).}{fig:ch05:sketch}
\ona{The fourth point needs a word. The minimum of a real function is a real number which may have no finite representation, and the same holds for the parameters that attain it. Following the usual convention, a minimisation problem counts as solved if it is solved to exponential precision, that is, up to an error $2^{-\mathrm{poly}}$ in the size of the input. The hardness results below therefore do not come from the difficulty of writing the minimum down.}
\end{frame}

\begin{frame}\ft{Training with a quantum computer: the oracular problem}
\onp{\footnotesize}
\begin{probl}[VQA minimisation, oracular formulation]\label{p:ch05:oracle}
\begin{description}\setlength\itemsep{0.1em}
    \item[Instance] Generators $H_1,\dots,H_L$ and an observable $O$ on $N$ qubits, each given as a real combination of polynomially many Pauli strings.
    \item[Oracle] For $\ket{\Psi(\boldsymbol\phi)} = U_L(\phi_L)\cdots U_1(\phi_1)\ket{0}^{\otimes N}$ with $U_i(\phi) = e^{-i\phi H_i}$, an oracle returns $\braket{O(\boldsymbol\phi)} = \braket{\Psi(\boldsymbol\phi)|O|\Psi(\boldsymbol\phi)}$ for any $\boldsymbol\phi$ up to any desired additive error $1/\mathrm{poly}$.
    \item[Task] Find $\boldsymbol\phi\in\R^L$ minimising $\braket{O(\boldsymbol\phi)}$.
\end{description}
\end{probl}
\ona{The oracle stands for the quantum computer, idealised in two ways: it is deterministic, as if one post-selected on the runs that give a correct estimate, and its precision is whatever the algorithm asks for, as long as it is inverse-polynomial, which polynomially many shots achieve. The noise that this idealisation hides is the subject of \Chap~\chref{C.IterationComplexity}. An algorithm for Problem~\ref{p:ch05:oracle} is a classical algorithm that may call the oracle, each call counting as one step. Whatever is hard for it is therefore hard because of the classical optimization alone; a better quantum computer would not help.}
\begin{propn}[\citet{bittel2021training}]\label{propn:ch05:oracle}
Unless $\mathbf P = \mathbf{NP}$, no deterministic classical algorithm solves \ona{Problem~\ref{p:ch05:oracle}}\onp{the oracular problem} in polynomial time.
\end{propn}
\ona{By itself this is not surprising. If the ansatz can reach every computational basis state and $O$ is a diagonal Ising Hamiltonian, minimising over the ansatz includes finding the ground state of $O$, which is $\mathbf{NP}$-hard (\Chap~\chref{C.Inapprox}). The interest lies in how little the proof needs, and in the variants below, where the ground-state problem is easy and the training is still hard.}
\end{frame}

\begin{frame}\ft{Training without a quantum computer}
\onp{\footnotesize}
\ona{If the Hilbert space has polynomial dimension $n$, that is, there are only $\mathcal O(\log n)$ qubits, then the state $\ket{\Psi(\boldsymbol\phi)}$ and its energy can be computed classically in polynomial time, and the oracle is unnecessary. It is then convenient to write everything in the computational basis.}
\begin{probl}[VQA minimisation]\label{p:ch05:vqa}
\begin{description}\setlength\itemsep{0.1em}
    \item[Instance] An initial state $\ket{\Psi_0}\in\C^n$, generators $H_1,\dots,H_L\in\Herm(\C^n)$ and an observable $O\in\Herm(\C^n)$, given as vectors and matrices.
    \item[Task] Find $\boldsymbol\phi\in\R^L$ minimising $\braket{O(\boldsymbol\phi)}$ for $\ket{\Psi(\boldsymbol\phi)} = U_L(\phi_L)\cdots U_1(\phi_1)\ket{\Psi_0}$.
\end{description}
\end{probl}
\ona{A matrix on $\lceil\log_2n\rceil$ qubits has at most $\mathcal O(n^2)$ Pauli coefficients, so this is a special case of the setting of Problem~\ref{p:ch05:oracle}, with a classically computable oracle. The \emph{decision version} asks, for a given threshold $a\in\R$, whether some $\boldsymbol\phi$ achieves $\braket{O(\boldsymbol\phi)}\le a$. It lies in $\mathbf{NP}$, with the convention on precision: the parameters are a certificate that can be checked in polynomial time. The decision version of Problem~\ref{p:ch05:oracle}, with a polynomial-time quantum computer in place of the oracle and an inverse-polynomial gap around $a$, lies in $\mathbf{QCMA}$, the class of problems whose ``yes'' instances have \emph{classical} certificates that a quantum computer can check \citep{bittel2021training,bittelQCMA}. This places the two kinds of difficulty side by side.}
\begin{itemize}\setlength\itemsep{0.1em}
    \item \textbf{The ansatz.} Deciding the ground energy of a local Hamiltonian is $\mathbf{QMA}$-complete \citep{kitaev2002classical,kempe2006complexity}, its certificate being a quantum state. If a polynomial-size ansatz always reached the ground energy, its parameters would be classical certificates; so if $\mathbf{QCMA}\ne\mathbf{QMA}$, every such ansatz fails on some Hamiltonians, whatever the optimizer does.
    \item \textbf{The optimizer.} Even when the ground-state problem is easy and the ground energy is within reach of the ansatz, finding the best parameters is $\mathbf{NP}$-hard, as we prove below.
\end{itemize}
\end{frame}

\begin{frame}\ft{QAOA as a training problem}
\onp{\footnotesize}
\begin{probl}[QAOA minimisation]\label{p:ch05:qaoa}
\begin{description}\setlength\itemsep{0.1em}
    \item[Instance] Hamiltonians $H_b,H_c\in\Herm(\C^n)$, and the number of layers $L$, written in unary.
    \item[Task] For $\ket{\Psi(\boldsymbol\beta,\boldsymbol\gamma)} = U_b(\beta_L)U_c(\gamma_L)\cdots U_b(\beta_1)U_c(\gamma_1)\ket{\Psi_0}$, where $\ket{\Psi_0}$ is the ground state of $H_b$, $U_b(\beta) = e^{-i\beta H_b}$ and $U_c(\gamma) = e^{-i\gamma H_c}$, find $\boldsymbol\beta,\boldsymbol\gamma\in\R^L$ minimising $\braket{H_c(\boldsymbol\beta,\boldsymbol\gamma)} = \braket{\Psi(\boldsymbol\beta,\boldsymbol\gamma)|H_c|\Psi(\boldsymbol\beta,\boldsymbol\gamma)}$.
\end{description}
\end{probl}
\ona{Three features distinguish this problem from Problem~\ref{p:ch05:vqa}: there are only two generators, used alternately, as in the discretised annealing of \Chap~\chref{C.Annealing}; the observable is one of them; and the initial state is not free but the ground state of $H_b$. QAOA for MaxCut (\Chap~\chref{C.QAOA}) is the case $H_b = -\sum_jX_j$, whose ground state is $\ket+^{\otimes n}$, and $H_c = H_C$, the diagonal cost Hamiltonian. The hardness proofs below use small, explicitly given matrices, and so they apply to any formulation that contains those.}
\begin{itemize}\setlength\itemsep{0.15em}
    \item \textbf{Why unary?} With $L$ written in binary, an input of $\mathcal O(\log L)$ bits could ask for $L$ layers, and evaluating a single energy would take time exponential in the size of the input. In unary, the circuit is no longer than its description.
    \item \textbf{Periodic angles.} If the eigenvalues of $H_b$ and $H_c$ are integers, the energy is $2\pi$-periodic in every angle, and one may restrict to $\boldsymbol\beta,\boldsymbol\gamma\in[0,2\pi)^L$, as for MaxCut.
\end{itemize}
\end{frame}

\begin{frame}\ft{How far from optimal? Model mismatch and optimization error}
\onp{\footnotesize}

The \emph{approximation error} of a run splits into two parts (Fig.~\ref{fig:ch05:errors}):}
\begin{equation}\label{eq:ch05:split}
    \delta = \frac{\braket{O}_a - \lambda_{\min}(O)}{w(O)}
    = \underbrace{\frac{\braket{O}_{\min} - \lambda_{\min}(O)}{w(O)}}_{\delta_m\text{: model mismatch}}
    + \underbrace{\frac{\braket{O}_a - \braket{O}_{\min}}{w(O)}}_{\delta_o\text{: optimization error}}.
\end{equation}

\ona{\caption{The two parts of the error of a variational algorithm, both measured in units of the spectral width. The model mismatch is a property of the ansatz; the optimization error is what the optimizer leaves, and is the subject of the hardness results.}\label{fig:ch05:errors}\end{figure}}
\begin{defn}[Optimization error, \citet{bittel2021training}]\label{defn:ch05:opterr}
The \emph{optimization error} of an algorithm is the smallest $\Delta\in[0,1]$ such that $\delta_o\le\Delta$ on every instance considered.
\end{defn}
\ona{Three remarks. (i) The normalisation puts $\delta$, $\delta_m$ and $\delta_o$ in $[0,1]$ and makes them invariant under $O\mapsto aO + b\one$ with $a>0$, so that neither the unit of energy nor a constant shift matters. (ii) $\Delta = 1$ is trivial, since any parameters whatsoever achieve it; a theorem that no polynomial-time algorithm has $\Delta<1$ says that nothing at all can be guaranteed. (iii) Like an approximation ratio (\Chap~\chref{C.Inapprox}), $\Delta$ is a worst case over all instances of a class, here a class of VQAs. When $\lambda_{\max}(O) = 0$ and the ansatz contains a ground state, as in step 2 below, an output within a ratio $\alpha$ of the optimum has $\delta_o = 1-\alpha$.}
\onp{$\Delta = 1$: trivial. $\Delta$ is to training what $1-\alpha$ is to approximation.}
\end{frame}

\section{Hardness of training: Bittel and Kliesch}\label{sec:ch05:hardness}

\begin{frame}\ft{Training a variational algorithm is $\mathbf{NP}$-hard}
\onp{\footnotesize}
\ona{Throughout, $A\in\{0,1\}^{d\times d}$ is the adjacency matrix of a simple, undirected, unweighted graph with $d$ vertices and $|E|\ge1$ edges, so that $A$ is symmetric with zero diagonal and $\sum_{ij}A_{ij} = 2|E|$, and $\mathrm{MC} = \mathrm{MaxCut}(A)$. The main results of \citet{bittel2021training}, in the terms just introduced, are the following.}
\begin{thm}[\citet{bittel2021training}]\label{thm:ch05:bittel}
The following are $\mathbf{NP}$-hard:
\begin{enumerate}\setlength\itemsep{0.05em}
    \item[(a)] \ona{Problem~\ref{p:ch05:oracle}}\onp{the oracular problem}, with single-qubit rotations on $d$ qubits;
    \item[(b)] \ona{Problem~\ref{p:ch05:vqa}}\onp{VQA minimisation} in dimension $2d$, that is on $\mathcal O(\log d)$ qubits, with $d$ layers or with one; its decision version is $\mathbf{NP}$-complete;
    \item[(c)] \ona{Problem~\ref{p:ch05:qaoa}}\onp{QAOA minimisation} with one layer;
    \item[(d)] \ona{Problem~\ref{p:ch05:qaoa}}\onp{QAOA minimisation} with $d$ layers, angles in $[0,2\pi)$, and $\|H_b\|= 3$, $\|H_c\| = 1$;
    \item[(e)] the analogue of \ona{Problem~\ref{p:ch05:vqa}}\onp{VQA minimisation} for free fermions.
\end{enumerate}
Moreover, unless $\mathbf P = \mathbf{NP}$, no polynomial-time algorithm has optimization error $\Delta<1$ for (a), nor $\Delta < (1-\alpha_{\max})/2$ for (d), where $\alpha_{\max}$ is the best approximation ratio for MaxCut achievable in polynomial time.
\end{thm}
\ona{In (b), (c) and (e) all energies can be computed classically in polynomial time: the hardness is intrinsic to the optimization, and not inherited from a hard ground-state problem or a hard simulation. The proof is a chain of reductions, all starting from MaxCut (Fig.~\ref{fig:ch05:chain}). We give all of them in full except the construction behind (d), of which we explain the idea.}
\end{frame}

\begin{frame}\ft{Step 1: from MaxCut to cosines}
\onp{\scriptsize}
\begin{probl}[Continuous MaxCut]\label{p:ch05:cmc}
Given $A$, find $\boldsymbol\phi\in[0,2\pi)^d$ minimising
\begin{equation}\label{eq:ch05:mu}
    \mu(\boldsymbol\phi) \;=\; \tfrac14\sum_{i,j}A_{ij}\,\big(\cos\phi_i\cos\phi_j - 1\big).
\end{equation}
\end{probl}
\begin{lem}[\citet{bittel2021training}]\label{lem:ch05:cmc}
$\min_{\boldsymbol\phi}\mu = -\mathrm{MC}$, and from any $\boldsymbol\phi$ one obtains in polynomial time a cut of value at least $-\mu(\boldsymbol\phi)$. Hence \ona{Problem~\ref{p:ch05:cmc}}\onp{continuous MaxCut} is $\mathbf{NP}$-hard, and unless $\mathbf P = \mathbf{NP}$ no polynomial-time algorithm approximates it within a ratio better than $\alpha_{\max}$.
\end{lem}
\ona{\emph{Proof.} Fix all coordinates but $\phi_i$. Since $A$ is symmetric with zero diagonal,}
\begin{equation}
    \mu(\boldsymbol\phi) \;=\; \cos\phi_i\,\Big[\tfrac12\sum_{j\ne i}A_{ij}\cos\phi_j\Big] \;+\; C_i,
\end{equation}
\ona{with $C_i$ independent of $\phi_i$. As a function of $\phi_i$, then, $\mu$ is affine in $\cos\phi_i \in [-1,1]$, and is smallest at $\phi_i = 0$ or $\phi_i = \pi$. Moving the coordinates one at a time to whichever of $0, \pi$ is better never increases $\mu$, and after $d$ steps $\boldsymbol\phi \in \{0,\pi\}^d$. There, $v_i = \cos\phi_i \in \{\pm1\}$ and}
\begin{equation}
    \mu \;=\; \tfrac14\sum_{i,j}A_{ij}(v_iv_j - 1) \;=\; -\tfrac12\sum_{i,j}A_{ij}\,\mathbb 1[v_i \ne v_j] \;=\; -\,\mathrm{cut}(v). \qquad\square
\end{equation}
\ona{The reduction preserves values exactly, so continuous MaxCut is exactly as hard to approximate as MaxCut. Note that the rounding is free: a continuous answer is never worse than the discrete one read off from it.}
\end{frame}

\begin{frame}\ft{Local minima are the local optima of MaxCut}
\onp{\footnotesize}
\ona{The proof of Lemma~\ref{lem:ch05:cmc} says more about the landscape than where its minimum is. The derivatives of $\mu$ are}
\begin{equation}
    \frac{\partial\mu}{\partial\phi_i} = -\tfrac12\sin\phi_i\sum_{j}A_{ij}\cos\phi_j,
    \qquad
    \frac{\partial^2\mu}{\partial\phi_i\,\partial\phi_k} = -\tfrac12\,\delta_{ik}\cos\phi_i\sum_jA_{ij}\cos\phi_j + \tfrac12A_{ik}\sin\phi_i\sin\phi_k.
\end{equation}
\ona{At a point of $\{0,\pi\}^d$, with $v_i = \cos\phi_i\in\{\pm1\}$, the gradient vanishes and the Hessian is diagonal, with entries $-\frac12v_i\sum_jA_{ij}v_j$: half the number of neighbours of $i$ on the other side of the cut minus half the number on its own side. The entry is positive exactly when moving $i$ to the other side would decrease the cut. Therefore:}
\begin{itemize}\setlength\itemsep{0.15em}
    \item the points of $\{0,\pi\}^d$ that are strict local minima of $\mu$ are the cuts that every single-vertex move makes worse, the fixed points of greedy local search;
    \item such a cut cuts at least half of the edges at every vertex, hence at least $|E|/2\ge\mathrm{MC}/2$ edges, and the bound $1/2$ is tight \ona{(Exercise~\ref{exe:ch05:persistent})}\onp{(exercise)};
    \item a gradient method, or any method that converges to a local minimum, such as Newton's method, BFGS or the natural gradient, started in the basin of such a minimum returns a cut of half the optimum: its ratio is $1/2$, against $0.878$ for Goemans--Williamson.
\end{itemize}
\ona{\citet{bittel2021training} call these minima \emph{persistent}. They are not artefacts of a poor parametrisation that a better ansatz could remove, but the local optima of the combinatorial problem itself, embedded in the landscape. Which of them a local method finds is decided by where it starts (Figs.~\ref{fig:ch05:persistent} and~\ref{fig:ch05:localminima}), which is the subject of \Chaps~\chref{C.WSRounded}--\chref{C.WSContinuous}.}
\end{frame}

\begin{frame}\ft{Step 2: a circuit that computes $\mu$}
\onp{\footnotesize}
\ona{\emph{Proof of Proposition~\ref{propn:ch05:oracle}.} We reduce Problem~\ref{p:ch05:cmc} to Problem~\ref{p:ch05:oracle}. Take $N = d$ qubits and $L = d$ layers with generators $H_i = \frac12Y_i$, so that $U_i(\phi_i) = R_Y(\phi_i)$ acts on qubit $i$, and the Ising observable}
\begin{equation}
    O \;=\; \tfrac14\sum_{i,j}A_{ij}\big(Z_iZ_j - \one\big).
\end{equation}
\ona{The state is a product, $\bigotimes_i\big(\cos\tfrac{\phi_i}2\ket0 + \sin\tfrac{\phi_i}2\ket1\big)$, with $\braket{Z_i} = \cos\phi_i$ and $\braket{Z_iZ_j} = \cos\phi_i\cos\phi_j$ for $i\ne j$, so that}
\begin{equation}
    \braket{O(\boldsymbol\phi)} \;=\; \mu(\boldsymbol\phi) \qquad\text{exactly.}
\end{equation}
\ona{The instance has $d$ generators and $\mathcal O(d^2)$ Pauli strings, and the oracle can be simulated, since $\mu(\boldsymbol\phi)$ is a sum of $\mathcal O(d^2)$ products of cosines. A polynomial-time algorithm for Problem~\ref{p:ch05:oracle}, with the oracle replaced by this computation and the rounding of Lemma~\ref{lem:ch05:cmc} appended, would solve MaxCut in polynomial time. \qed}
\begin{itemize}\setlength\itemsep{0.15em}
    \item The circuit has no entanglement at all, and the ground state of $O$, a computational basis state, lies in the ansatz: $\braket{O}_{\min} = \lambda_{\min}(O) = -\mathrm{MC}$ and $\delta_m = 0$. The hardness lies wholly in the classical optimization.
    \item The eigenvalues of $O$ are $-\mathrm{cut}(z)$, $z\in\{0,1\}^d$, so $w(O) = \mathrm{MC}$, and a training algorithm with $\Delta$ is a MaxCut algorithm of ratio $1-\Delta$. Unless $\mathbf P = \mathbf{NP}$, therefore, $\Delta\ge1-\alpha_{\max}$.
\end{itemize}
\end{frame}

\begin{frame}\ft{Boosting: no guarantee at all}
\onp{\footnotesize}
\begin{cor}[\citet{bittel2021training}]\label{cor:ch05:boost}
Unless $\mathbf P = \mathbf{NP}$, no polynomial-time algorithm for \ona{Problem~\ref{p:ch05:oracle}}\onp{the oracular problem} has optimization error $\Delta<1$.
\end{cor}
\ona{\emph{Proof.} Take $k$ copies of the circuit of step 2, on $kd$ qubits, all driven by the same parameters, $\tilde U(\boldsymbol\phi) = U(\boldsymbol\phi)^{\otimes k}$, and the observable}
\begin{equation}
    \tilde O = (-1)^{k-1}O^{\otimes k}, \qquad \braket{\tilde O(\boldsymbol\phi)} = (-1)^{k-1}\mu(\boldsymbol\phi)^k = -|\mu(\boldsymbol\phi)|^k,
\end{equation}
\ona{using $\mu\le0$; the sign keeps the problem a minimisation. For constant $k$, $\tilde O$ has $\mathcal O(d^{2k})$ Pauli strings. Its eigenvalues are $(-1)^{k-1}$ times products of $k$ eigenvalues of $O$, which lie in $[-\mathrm{MC},0]$, so they lie in $[-\mathrm{MC}^k,0]$, and $w(\tilde O) = \mathrm{MC}^k = -\braket{\tilde O}_{\min}$. Suppose an algorithm guarantees $\Delta\le\rho<1$. On this instance it returns $\boldsymbol\phi$ with}
\begin{equation}
    \mathrm{MC}^k - |\mu(\boldsymbol\phi)|^k \;\le\; \rho\,\mathrm{MC}^k, \qquad\text{that is,}\qquad |\mu(\boldsymbol\phi)| \;\ge\; (1-\rho)^{1/k}\,\mathrm{MC}.
\end{equation}
\ona{Choose $k$ so large that $(1-\rho)^{1/k} > 16/17$; for $\rho = 0.9$, $k = 38$ suffices. Rounding $\boldsymbol\phi$ as in Lemma~\ref{lem:ch05:cmc} then gives, in polynomial time, a cut within a factor $16/17$ of the optimum, which is $\mathbf{NP}$-hard \citep{Hstad2001}. \qed}
\ona{The $k$-th power stretches the scale: a relative error $\epsilon$ in $|\mu|$ becomes $1-(1-\epsilon)^k\approx k\epsilon$ in $|\mu|^k$. On general VQAs, the worst-case guarantee of a training algorithm can therefore be made as poor as one likes.}
\onp{$k$ copies: error $\rho$ gives a cut of ratio $(1-\rho)^{1/k}\to1$; beating $16/17$ is $\mathbf{NP}$-hard.}
\end{frame}

\begin{frame}\ft{Step 3: logarithmically many qubits}
\onp{\scriptsize}
\ona{The circuit of step 2 acts on $d$ qubits, and only its special structure makes its energy easy to compute. To show that the hardness does not depend on this, \citet{bittel2021training} encode Problem~\ref{p:ch05:cmc} in a Hilbert space of dimension $2d$, where everything is a small matrix computation.}
\begin{thm}[\citet{bittel2021training}]\label{thm:ch05:log}
\ona{Problem~\ref{p:ch05:vqa}}\onp{VQA minimisation} is $\mathbf{NP}$-hard already in dimension $n = 2d$ with $L = d$ layers, and its decision version is $\mathbf{NP}$-complete.
\end{thm}
\ona{\emph{Proof.} Take the uniform initial state $\ket{\Psi_0} = \ket{+_{2d}} = (2d)^{-1/2}\sum_{k=1}^{2d}\ket k$ and the diagonal generators $H_j = \ket{2j-1}\!\bra{2j-1} - \ket{2j}\!\bra{2j}$, which imprint opposite phases on the $j$-th pair of basis states:}
\begin{equation}
    \ket{\Psi(\boldsymbol\phi)} = \frac1{\sqrt{2d}}\sum_{j=1}^d\big(e^{-i\phi_j}\ket{2j-1} + e^{i\phi_j}\ket{2j}\big).
\end{equation}
\ona{For the observable, double every vertex: let $W = A\otimes J_2$, with $J_2 = \begin{psmallmatrix}1&1\\1&1\end{psmallmatrix}$, be the adjacency matrix of the graph on $2d$ vertices in which both copies of $i$ are joined to both copies of $j$ whenever $ij$ is an edge, let $L_W = \mathrm{diag}(W\mathbf 1) - W$ be its Laplacian, and}
\begin{equation}
    O \;=\; -\tfrac d8\,L_W \;=\; \tfrac d8\,A\otimes J_2 \;-\; \tfrac d4\,\mathrm{diag}(A\mathbf 1)\otimes\one_2.
\end{equation}
\ona{This is the observable of \citet{bittel2021training}, who write it as $O' = \frac d8A\otimes J_2$ with the diagonal replaced by minus the column sums. A Laplacian satisfies $x^\dagger L_Wx = \frac12\sum_{k,l}W_{kl}|x_k-x_l|^2$. The amplitudes of $\ket{\Psi(\boldsymbol\phi)}$ are $(2d)^{-1/2}e^{-is\phi_j}$ with $s = \pm1$, and for an edge $ij$ the four pairs of signs give $\sum_{s,s'}|e^{-is\phi_i} - e^{-is'\phi_j}|^2 = 8 - 4\cos(\phi_i-\phi_j) - 4\cos(\phi_i+\phi_j)$, so that}
\begin{equation}
    \braket{O(\boldsymbol\phi)} = \tfrac18\sum_{i,j}A_{ij}\big(\cos(\phi_i-\phi_j) + \cos(\phi_i+\phi_j) - 2\big) = \mu(\boldsymbol\phi).
\end{equation}
\ona{This reduces Problem~\ref{p:ch05:cmc} to Problem~\ref{p:ch05:vqa}. For the decision version, a parameter vector is a certificate that can be checked in polynomial time. \qed}
\onp{$O = -\frac d8L_W$, minus a Laplacian of the doubled graph; $\braket{O(\boldsymbol\phi)} = \mu(\boldsymbol\phi)$.}
\end{frame}

\begin{frame}\ft{Here the ground-state problem is easy, and training is not}
\onp{\footnotesize}
\ona{The construction of step 3 separates the two parts of the error \eqref{eq:ch05:split} sharply. Since $O = -\frac d8L_W$ and a Laplacian is positive semidefinite with the constant vector in its kernel, $\lambda_{\max}(O) = 0$ and $\lambda_{\min}(O) = -\frac d8\lambda_{\max}(L_W)$. The vectors $v\otimes(1,1)$ and $v\otimes(1,-1)$ show that the spectrum of $L_W$ consists of twice the eigenvalues of the Laplacian $L_A = \mathrm{diag}(A\mathbf 1) - A$ of the graph and twice its degrees; as $\lambda_{\max}(L_A)\ge e_i^\top L_Ae_i = \deg(i)$, the largest is $2\lambda_{\max}(L_A)$. Hence}
\begin{equation}
    \lambda_{\min}(O) \;=\; -\tfrac d4\,\lambda_{\max}(L_A) \;\le\; -\mathrm{MC} \;=\; \braket{O}_{\min},
\end{equation}
\ona{where the inequality is the eigenvalue bound on MaxCut of \citet{mohar1990eigenvalues}: for $v\in\{\pm1\}^d$, $\mathrm{cut}(v) = \frac14v^\top L_Av\le\frac d4\lambda_{\max}(L_A)$.}
\begin{itemize}\setlength\itemsep{0.15em}
    \item \textbf{The ground-state problem} of $O$ is the largest eigenvalue of a $d\times d$ matrix, computable in polynomial time; its value is an upper bound on MaxCut, which is usually not attained.
    \item \textbf{The ansatz} reaches only states with amplitudes of equal modulus and opposite phases in each pair. Its best energy is $-\mathrm{MC}$, so the model mismatch $\delta_m$ is the gap between the eigenvalue bound and the maximum cut, relative to the bound.
    \item \textbf{The optimization} within the ansatz is $\mathbf{NP}$-hard. Restricting the states made the problem harder, not easier.
\end{itemize}
\end{frame}

\begin{frame}\ft{Step 4: a clock with $d$ hands}
\onp{\footnotesize}
\ona{With $d$ layers, each parameter sets one phase. A single layer has a single parameter, a time $t$, and $e^{-itH}$ with $H = \sum_jE_j(\ket{2j-1}\!\bra{2j-1} - \ket{2j}\!\bra{2j})$ sets all the phases at once, to $\phi_j = E_jt$. The question is whether the curve $t\mapsto\boldsymbol Et$ comes close to every point of the torus $[0,2\pi)^d$, for energies with short descriptions.}
\begin{defn}[\citet{bittel2021training}]
Energies $E_1,\dots,E_d$ form an \emph{$\epsilon$-approximate ergodic spectrum} if for every $\boldsymbol\phi\in[0,2\pi)^d$ there is a $t\ge0$ with $|\phi_j - E_jt|_{2\pi}\le\epsilon$ for all $j$, where $|x|_{2\pi} = \min_{k\in\mathbb Z}|x-2\pi k|$.
\end{defn}
\ona{Energies that are linearly independent over the rationals are $\epsilon$-approximate ergodic for every $\epsilon>0$, by Kronecker's theorem, but the times needed may be huge and the energies cannot be written down exactly. A clock does better.}
\begin{lem}[\citet{bittel2021training}]\label{lem:ch05:clock}
For an integer $m\ge2$, the energies $E_j = 2\pi/m^j$, $j = 1,\dots,d$, form a $4\pi/m$-approximate ergodic spectrum, with integer times $0\le t<m^d$.
\end{lem}
\ona{\emph{Proof.} Write the target angles in base $m$: let $s_j = \lfloor m\phi_j/2\pi\rfloor\in\{0,\dots,m-1\}$, and let $t = \sum_{k=1}^ds_km^{k-1}$ be the number with digits $s_d\cdots s_2s_1$. Then}
\begin{equation}
    E_jt \;=\; 2\pi\sum_{k>j}s_km^{k-1-j} \;+\; \frac{2\pi s_j}m \;+\; 2\pi\sum_{k<j}s_km^{k-1-j}.
\end{equation}
\ona{The first sum is a multiple of $2\pi$. The middle term is within $2\pi/m$ of $\phi_j$, by the choice of $s_j$. The last sum is at most $2\pi(m-1)\sum_{k<j}m^{k-1-j}<2\pi/m$. Hence $|\phi_j - E_jt|_{2\pi}<4\pi/m$. \qed}
\ona{Hand $j$ turns once in $m^j$ units of time, $m$ times more slowly than hand $j-1$. For $m = 10$ it points, up to a small contribution of the lower digits, at the $j$-th digit of $t$ from the right, as on an odometer (Exercise~\ref{exe:ch05:clock}).}
\onp{Digits of $t$ in base $m$ set the hands; error $<2\pi/m$ from rounding plus $<2\pi/m$ from the lower digits.}
\end{frame}

\begin{frame}\ft{One layer: the reduction}
\onp{\footnotesize}
\begin{thm}[\citet{bittel2021training}]\label{thm:ch05:onelayer}
\ona{Problem~\ref{p:ch05:vqa}}\onp{VQA minimisation} is $\mathbf{NP}$-hard with a single layer, $L = 1$.
\end{thm}
\ona{\emph{Proof.} Keep $\ket{\Psi_0}$ and $O$ from step 3, and use the single generator $H = \sum_jE_j(\ket{2j-1}\!\bra{2j-1} - \ket{2j}\!\bra{2j})$ with the energies of Lemma~\ref{lem:ch05:clock}. Then $\braket{O(t)} = \mu(\boldsymbol Et)$. How fine must the clock be? Since $|\partial\mu/\partial\phi_j|\le\frac12\deg(j)$ and the degrees sum to $2|E|$,}
\begin{equation}
    |\mu(\boldsymbol\phi) - \mu(\boldsymbol\phi')| \;\le\; |E|\,\max_j|\phi_j - \phi'_j|_{2\pi}.
\end{equation}
\ona{Taking for $\boldsymbol\phi$ a maximum cut and for $t$ the time of the lemma gives $\min_t\braket{O(t)}\le-\mathrm{MC} + 4\pi|E|/m$. If an algorithm returns a $t$ within $\eta$ of the minimum, rounding $\boldsymbol Et$ as in Lemma~\ref{lem:ch05:cmc} gives a cut of value greater than $\mathrm{MC} - 4\pi|E|/m - \eta$. When $m>8\pi|E|$ and $\eta<1/2$, this is a maximum cut, since cut values are integers. The energies then need $\mathcal O(d\log|E|)$ bits; after rescaling $t$ by $2\pi$, they are the rationals $m^{-j}$. \qed}
\begin{cor}[\citet{bittel2021training}]\label{cor:ch05:time}
For a time-independent Hamiltonian on logarithmically many qubits, minimising $\braket{O(t)}$ over the time $t\ge0$ of the evolution is $\mathbf{NP}$-hard.
\end{cor}
\ona{The single parameter does not make the landscape simple: as a function of $t$, $\mu(\boldsymbol Et)$ is a sum of cosines with frequencies spanning $d$ orders of magnitude, and it has exponentially many local minima (Fig.~\ref{fig:ch05:onelayer}).}
\end{frame}

\begin{frame}\ft{Step 5: QAOA with one layer}
\onp{\scriptsize}
\begin{thm}[\citet{bittel2021training}]\label{thm:ch05:qaoa1}
\ona{Problem~\ref{p:ch05:qaoa}}\onp{QAOA minimisation} is $\mathbf{NP}$-hard with a single layer, $L = 1$.
\end{thm}
\ona{\emph{Proof.} QAOA differs from step 4 in that the initial state must be the ground state of $H_b$ and the observable must be $H_c$. Both are arranged by adding one level. On $\C^{2d+1}$ let}
\begin{equation}
    H_b = \mathrm{diag}(E_1,-E_1,\dots,E_d,-E_d,-1), \qquad H_c = O\oplus0 + \tau\big(\ket{+_{2d}}\!\bra{2d+1} + \ket{2d+1}\!\bra{+_{2d}}\big),
\end{equation}
\ona{with $O$ from step 3, the energies of Lemma~\ref{lem:ch05:clock} with $m\ge7$, so that $|E_j|<1$, and a coupling $\tau>0$. The ground state of $H_b$ is the extra level $\ket{2d+1}$. As $O\ket{+_{2d}} = 0$, the uniform vector being in the kernel of a Laplacian, $H_c$ acts on the span of $\ket{2d+1}$ and $\ket{+_{2d}}$ as $\tau$ times a Pauli $X$, and}
\begin{equation}
    U_c(\gamma)\ket{2d+1} = \cos(\tau\gamma)\ket{2d+1} - i\sin(\tau\gamma)\ket{+_{2d}}.
\end{equation}
\ona{The mixer $U_b(\beta)$ then imprints the phases $e^{\mp iE_j\beta}$ on the pairs, as the single layer of step 4 does, and $e^{i\beta}$ on the extra level. A short computation (Exercise~\ref{exe:ch05:qaoa}) gives}
\begin{equation}
    \braket{H_c(\beta,\gamma)} = \sin^2(\tau\gamma)\,\mu(\boldsymbol E\beta) + \tau\sin(2\tau\gamma)\,g(\beta), \qquad g(\beta) = -\frac{\sin\beta}d\sum_{j=1}^d\cos(E_j\beta),
\end{equation}
\ona{with $|g|\le1$. Since $\mu\le0$, every $(\beta,\gamma)$ has $\braket{H_c}\ge\mu(\boldsymbol E\beta) - \tau$, while $\gamma = \pi/(2\tau)$ gives $\braket{H_c} = \mu(\boldsymbol E\beta)$ exactly. Angles within $\eta$ of the optimum therefore yield a $\beta$ with $\mu(\boldsymbol E\beta)\le-\mathrm{MC} + 4\pi|E|/m + \tau + \eta$, and with $\tau = 1/4$, $\eta<1/4$ and $m>8\pi|E|$ the rounding of step 4 recovers a maximum cut. \qed}
\onp{One extra level, the ground state of $H_b$, weakly coupled to $\ket{+_{2d}}$; $\gamma = \pi/2\tau$ moves the state into the construction of step 4.}
\end{frame}

\begin{frame}\ft{Step 6: QAOA with bounded energies}
\onp{\scriptsize}
\ona{The energies $2\pi/m^j$ of step 5 span $d$ orders of magnitude, and a device implementing them would need $d\log m$ bits of precision. With $d$ layers, small integer spectra suffice.}
\begin{thm}[\citet{bittel2021training}]\label{thm:ch05:qaoad}
\ona{Problem~\ref{p:ch05:qaoa}}\onp{QAOA minimisation} is $\mathbf{NP}$-hard with $L = d$ layers and angles in $[0,2\pi)$, even for $H_b$ and $H_c$ with integer eigenvalues, $\|H_b\| = 3$ and $\|H_c\| = 1$, and with $\braket{H_c}_{\min} = 1 - 2\,\mathrm{MC}/|E|$.
\end{thm}
\ona{\emph{Idea of the construction.} The Hilbert space is the direct sum of $2d+1$ copies of $\C^d\otimes\C^d\otimes\C^2\otimes\C^2$, with basis states $\ket{i,j,a,b}$: an ordered pair of vertices and two bits. The ground state of $H_b$, which is the initial state, is the uniform superposition over the edges $ij$ and the bits, in the first copy. Each half-layer moves amplitude one copy further: $U_c(\gamma_k)$ with amplitude $\sin(\gamma_k/2)$, $U_b(\beta_k)$ with amplitude $\sin\beta_k$, and the latter also attaches the phase $e^{ia(\beta_k+\pi/2)}$ to the components with $i = k$, and $e^{(-1)^bia(\beta_k+\pi/2)}$ to those with $j = k$. After all $d$ layers, the last copy carries the phases of all the $\beta_k$, and there $H_c$ measures their interference. Amplitude left behind in the other copies only adds energy, as $H_c$ is positive semidefinite there. The result is}
\begin{equation}
    \braket{H_c(\boldsymbol\beta,\boldsymbol\gamma)} \;\ge\; \prod_{k=1}^d\sin^2\beta_k\,\sin^2\frac{\gamma_k}2\;\cdot\;\frac1{2|E|}\sum_{i,j}A_{ij}\sin\beta_i\sin\beta_j,
\end{equation}
\ona{with equality for $\gamma_k = \pi$ and $\boldsymbol\beta\in\{\pi/2,3\pi/2\}^d$. Since $\sum_{ij}A_{ij}\sin\beta_i\sin\beta_j = 4\mu(\boldsymbol\beta-\pi/2) + 2|E|$, the minimum is $1 - 2\,\mathrm{MC}/|E|$. From any angles, a cut of value at least $\frac{|E|}2\big(1-\braket{H_c}\big)$ is obtained by rounding $\boldsymbol\beta-\pi/2$ as in Lemma~\ref{lem:ch05:cmc}, or, if that is worse, by greedy local search, which cuts at least $|E|/2$ edges.}
\onp{$2d+1$ copies of a register; each half-layer moves amplitude one copy on and imprints the phase of $\beta_k$ on vertex $k$; the energy is read in the last copy.}
\end{frame}

\begin{frame}\ft{No guarantee below $(1-\alpha_{\max})/2$}
\onp{\footnotesize}
\begin{cor}[\citet{bittel2021training}]\label{cor:ch05:qaoa}
Unless $\mathbf P = \mathbf{NP}$, every polynomial-time algorithm for \ona{Problem~\ref{p:ch05:qaoa}}\onp{QAOA minimisation}, and hence for \ona{Problem~\ref{p:ch05:vqa}}\onp{VQA minimisation}, has optimization error $\Delta\ge(1-\alpha_{\max})/2$.
\end{cor}
\ona{\emph{Proof.} On the instances of Theorem~\ref{thm:ch05:qaoad}, $w(H_c) = 2$ and $\braket{H_c}_{\min} = 1-2\,\mathrm{MC}/|E|$. An algorithm with optimization error $\Delta$ returns angles with $\braket{H_c}_a\le1-2\,\mathrm{MC}/|E| + 2\Delta$, from which we read off a cut of value at least}
\begin{equation}
    \frac{|E|}2\big(1-\braket{H_c}_a\big) \;\ge\; \mathrm{MC} - \Delta|E| \;\ge\; (1-2\Delta)\,\mathrm{MC},
\end{equation}
\ona{using $\mathrm{MC}\ge|E|/2$. This is a MaxCut algorithm of ratio $1-2\Delta$, so $1-2\Delta\le\alpha_{\max}$. \qed}
\begin{itemize}\setlength\itemsep{0.2em}
    \item \textbf{The numbers.} With H\accent23 astad's $\alpha_{\max}\le16/17$ \citep{Hstad2001}, $\Delta\ge1/34\approx0.029$. Under the Unique Games Conjecture, $\alpha_{\max} = \alpha_{\rm GW}$ and $\Delta\ge0.061$.
    \item \textbf{Gradient methods.} A local minimum at a locally optimal cut of value $\mathrm{MC}/2$ has $\braket{H_c}_a - \braket{H_c}_{\min} = \mathrm{MC}/|E|\ge1/2$, so methods that may stop at local minima have $\Delta\ge1/4$, with no assumption on $\mathbf P$ and $\mathbf{NP}$.
    \item \textbf{No boosting.} The trick of \ona{Corollary~\ref{cor:ch05:boost}}\onp{boosting} does not apply: a tensor power of $H_c$ is not the cost Hamiltonian of a QAOA circuit with mixer $H_b$, and whether $\Delta$ can be pushed towards $1$ for QAOA is open.
\end{itemize}
\end{frame}

\begin{frame}\ft{Step 7: free fermions}
\onp{\scriptsize}
\ona{Free fermions are a second family of quantum systems that can be simulated classically, now with an exponentially large Hilbert space. Consider $n$ fermionic modes with annihilation operators $c_1,\dots,c_n$, $\{c_i^\dagger,c_j\} = \delta_{ij}$, $\{c_i,c_j\} = 0$. A \emph{quadratic} operator $H = \sum_{ij}h_{ij}c_i^\dagger c_j$ is described by its Hermitian \emph{coefficient matrix} $h$, and everything reduces to $n\times n$ matrices:}
\begin{itemize}\setlength\itemsep{0.1em}
    \item \textbf{evolution:} $e^{iHt}Oe^{-iHt}$ is quadratic, with coefficient matrix $e^{iht}oe^{-iht}$;
    \item \textbf{ground states:} the ground state $\rho[H_0]$ of a quadratic $H_0$ fills the modes of the negative eigenvalues of $h_0$, and its \emph{correlation matrix} $\Gamma_{ij} = \Tr(c_j^\dagger c_i\,\rho)$ is the projector onto their span;
    \item \textbf{expectations:} $\Tr(O\rho) = \sum_{ij}o_{ij}\Gamma_{ji} = \Tr(o\,\Gamma)$.
\end{itemize}
\begin{probl}[VQA minimisation, free fermions]\label{p:ch05:ff}
Given coefficient matrices $h_0,h_1,\dots,h_L,o\in\Herm(\C^n)$, find $\boldsymbol\phi\in\R^L$ minimising $\Tr\big(O\,U(\boldsymbol\phi)\,\rho[H_0]\,U(\boldsymbol\phi)^\dagger\big)$, where $U(\boldsymbol\phi) = e^{-i\phi_LH_L}\cdots e^{-i\phi_1H_1}$.
\end{probl}
\begin{thm}[\citet{bittel2021training}]\label{thm:ch05:ff}
\ona{Problem~\ref{p:ch05:ff}}\onp{The free-fermion problem} is $\mathbf{NP}$-hard, even when the initial state $\rho[H_0]$ is pure.
\end{thm}
\ona{\emph{Proof.} Take $n = 2d$ modes, $h_0 = \one - \frac2nJ$ with $J$ the all-ones matrix, $h_j = \ket{2j-1}\!\bra{2j-1} - \ket{2j}\!\bra{2j}$ as in step 3, and $o = O$, the matrix of step 3. The spectrum of $h_0$ is $(-1,1,\dots,1)$, with eigenvector $\ket{+_{2d}}$ for $-1$, so the ground state is pure: a single fermion in the mode $n^{-1/2}\sum_jc_j^\dagger$, with $\Gamma = \ket{+_{2d}}\!\bra{+_{2d}}$. The generators commute, and}
\begin{equation}
    \braket{O(\boldsymbol\phi)} = \Tr\big(o(\boldsymbol\phi)\,\Gamma\big) = \braket{+_{2d}|\,e^{i\sum_j\phi_jh_j}\,o\,e^{-i\sum_j\phi_jh_j}\,|+_{2d}} = \mu(\boldsymbol\phi),
\end{equation}
\ona{by the computation of step 3: a single fermion in $2d$ modes is a $2d$-level system. \qed{} (The published version has $h_0 = \one - J/n$, whose spectrum $(0,1,\dots,1)$ would give a degenerate ground state; the factor $2$ gives the spectrum stated there.)}
\onp{One fermion in $2d$ modes is a $2d$-level system: the free-fermion problem contains step 3.}
\end{frame}

\begin{frame}\ft{Training, if the Unique Games Theorem holds}
\onp{\scriptsize}
\ona{The bound of Corollary~\ref{cor:ch05:qaoa} is stated relative to $\alpha_{\max}$, the best ratio achievable for MaxCut in polynomial time, and so it sharpens if the Unique Games Theorem claimed in \Chap~\chref{C.Inapprox} is correct. Two cases must be distinguished, according to the computer on which the training runs.}
\begin{itemize}\setlength\itemsep{0.3em}
    \item \textbf{Classical training.} Goemans--Williamson achieves $\alpha_{\rm GW}$, and if the theorem holds and $\mathbf P\ne\mathbf{NP}$, no polynomial-time algorithm achieves more on every instance, so that $\alpha_{\max} = \alpha_{\rm GW}$. Every polynomial-time classical training procedure for the QAOA instances of \ona{Theorem~\ref{thm:ch05:qaoad}}\onp{step 6} then has optimization error
    \begin{equation*}
        \Delta \;\ge\; \tfrac12(1-\alpha_{\rm GW}) \;\approx\; 0.061,
    \end{equation*}
    with no assumption beyond $\mathbf P\ne\mathbf{NP}$. Without the theorem, the unconditional bound is $1/34 \approx 0.029$, from H\accent23 astad's $16/17$.
    \item \textbf{Training with a quantum computer in the loop.} An optimizer that queries a quantum device, or runs any other quantum subroutine, is a polynomial-time quantum algorithm. If it guaranteed $\Delta < (1-\alpha_{\rm GW})/2$, the reduction would turn it into a polynomial-time quantum algorithm approximating MaxCut beyond $\alpha_{\rm GW}$. By the theorem, that would decide 3-SAT, so the same bound holds unless $\mathbf{NP}\subseteq\mathbf{BQP}$. The quantum device does not help with the hard instances in any case: they live on $\mathcal O(\log d)$ qubits and can be simulated classically, so the difficulty lies in the search over parameters.
    \item \textbf{What is unchanged.} The oracle setting already excludes any guarantee $\Delta<1$ assuming only $\mathbf P\ne\mathbf{NP}$, or $\mathbf{NP}\not\subseteq\mathbf{BQP}$ for quantum optimizers. Local methods guarantee no more than $\Delta\ge1/4$ in any case. The $\mathbf{QCMA}$-hardness of approximating the optimal depth \citep{bittelQCMA} does not involve unique games.
\end{itemize}
\ona{In short, if the theorem holds, an error of about $6\%$ of the spectral width is the best that any efficient training procedure, classical or quantum, can guarantee on all instances of QAOA, unless $\mathbf{NP}\subseteq\mathbf{BQP}$. As before, this says nothing about typical instances.}
\end{frame}

\begin{frame}\ft{What the theorem says, and what it does not}
\begin{itemize}\setlength\itemsep{0.3em}
    \item It says that in the worst case no polynomial-time training procedure finds the optimal parameters, nor even parameters within a fixed fraction of the spectral width of the optimum, unless $\mathbf P = \mathbf{NP}$. The difficulty lies in the classical optimization, not in the quantum hardware or in estimating expectation values.
    \item It does not say that good parameters are hard to find for typical instances; \Chap~\chref{C.PerIteration} finds that they often are not.
    \item Nor does it say that local minima are useless: a warm start which inherits a classical guarantee keeps it whatever the optimizer does (\Chaps~\chref{C.WSRounded}--\chref{C.WSContinuous}).
\end{itemize}
\ona{Variational methods are in good company. The optimization in the Hartree--Fock method is $\mathbf{NP}$-hard \citep{schuch2009computational}, and so are global variations in the density matrix renormalisation group \citep{eisert2006computational}; both are used every day. Later work has extended the picture for VQAs. Approximating the depth that an ansatz needs to reach a target energy is $\mathbf{QCMA}$-hard \citep{bittelQCMA}; in an idealised digitised setting the training problem is even undecidable \citep{korpas2025undecidable}; and for a constant number of angles there are tractable regimes \citep{sakos2026global}.}
\onp{\footnotesize Also $\mathbf{NP}$-hard, and used daily: Hartree--Fock \citep{schuch2009computational}, DMRG \citep{eisert2006computational}. Later: $\mathbf{QCMA}$-hard depth \citep{bittelQCMA}; undecidability \citep{korpas2025undecidable}; tractable constant-angle regimes \citep{sakos2026global}.}
\end{frame}

\section{A second obstacle: barren plateaus}

\begin{frame}\ft{Barren plateaus}
\ona{Persistent local minima are one obstacle to training. A second, quite different one is the \emph{barren plateau} \citep{Clean_2018_BarrenPlateaus}: for sufficiently random, deep, problem-agnostic circuits, the gradient of $C$ has zero mean and a variance that decays exponentially in the number of qubits,}
\begin{equation}
    \mathrm{Var}\Big[\frac{\partial C}{\partial\vartheta_i}\Big] \in \mathcal O(2^{-n}),
\end{equation}
\ona{so that exponentially many shots are needed to see any gradient at all. Causes identified since: circuits close to unitary 2-designs, too much entanglement with partial measurements, global cost functions, noise; unified by the dimension of the dynamical Lie algebra of the generators. The landscapes of \citet{bittel2021training} have no barren plateaus, as they live on few qubits; their minima are hard to find for a combinatorial reason (Fig.~\ref{fig:ch05:obstacles}).}
\begin{itemize}\setlength\itemsep{0.2em}
    \item Problem-inspired ansätze with few parameters (QAOA, UCC) are less affected than hardware-efficient ones.
    \item A good initialisation that brings the state near the optimum avoids the plateau; how to find one is the question warm starts answer (\Chaps~\chref{C.WSRounded}--\chref{C.WSContinuous}).
\end{itemize}
\end{frame}

\ona{\section*{Exercises}}

\begin{frame}\ft{Exercises}
\onp{\scriptsize}
\begin{exe}[A first VQA]
Consider the problem of finding the lowest energy eigenstate of the spin-chain Hamiltonian
\begin{align*}
    H = J \sum_{i=1}^{n-1} Z_iZ_{i+1} + h \sum_{i=1}^n X_i,
\end{align*}
and let $J<0$. This is the \emph{transverse field Ising model}. Use the variational principle for the product ansatz state
\begin{align}
    \ket{\psi(\theta,\phi)} = \big(\cos(\theta)\ket{0} + e^{i \phi}\sin(\theta)\ket{1}\big)^{\otimes n}.
\end{align}
Then run a small experiment on (minimum) 3 qubits with small $h$. Compare and present the result. Should you be able to do this successfully, you will have computed your first VQA. The exact solution can be found using the Jordan--Wigner transform.
\end{exe}
\end{frame}

\begin{frame}\ft{Exercises, continued}
\onp{\footnotesize}
\begin{exe}[The error measure]
Show that $\delta$, $\delta_m$ and $\delta_o$ of \eqref{eq:ch05:split} lie in $[0,1]$ and are invariant under $O\mapsto aO+b\one$ for $a>0$. Compute the spectral width of the Ising observable of step 2, and of the observable of step 3, for the 4-cycle and for the triangle.
\end{exe}
\begin{exe}[Continuous MaxCut]
For the triangle, write out $\mu(\phi_1,\phi_2,\phi_3)$ from \eqref{eq:ch05:mu}, find its minimum, and show that the set of minimisers is not finite. Does the rounding of the proof still reach a maximum cut?
\end{exe}
\begin{exe}[Persistent local minima]\label{exe:ch05:persistent}
Let $V = L_1\cup L_2\cup R_1\cup R_2$ with four parts of $k$ vertices each, and join every vertex of $L_1\cup L_2$ to every vertex of $R_1\cup R_2$, except for a perfect matching between $L_1$ and $R_1$ and one between $L_2$ and $R_2$. Show that the cut $(L_1\cup R_1, L_2\cup R_2)$ corresponds to a strict local minimum of $\mu$, and that its ratio to the maximum cut is $k/(2k-1)\to1/2$.
\end{exe}
\end{frame}

\begin{frame}\ft{Exercises, continued}
\onp{\footnotesize}
\begin{exe}[Boosting]
Verify that the eigenvalues of $(-1)^{k-1}O^{\otimes k}$ lie in $[-\mathrm{MC}^k,0]$. Find the smallest $k$ for which an algorithm with $\Delta\le\rho$ would beat H\accent23 astad's $16/17$, for $\rho = 0.9$ and for $\rho = 0.99$.
\end{exe}
\begin{exe}[A clock]\label{exe:ch05:clock}
Let $m = 10$, $d = 3$ and $\boldsymbol\phi = 2\pi\,(0.37, 0.81, 0.25)$. Compute the digits $s_j$ and the time $t$ of \ona{Lemma~\ref{lem:ch05:clock}}\onp{the clock lemma}, and the errors $|\phi_j - E_jt|_{2\pi}$; compare them with $4\pi/m$. Why does the error of hand $j$ depend only on $\phi_j$ and on the digits $s_k$ with $k<j$?
\end{exe}
\end{frame}

\begin{frame}\ft{Exercises, continued}
\onp{\footnotesize}
\begin{exe}[QAOA with one layer]\label{exe:ch05:qaoa}

\begin{exe}[Free fermions]
Show that the quadratic Hamiltonian with $h_0 = \one - \frac2nJ$ has a unique ground state, with one fermion and correlation matrix $J/n$, and that with $h_0 = \one - \frac1nJ$ the ground state is degenerate. What is the correlation matrix of the maximally mixed state on the ground space in the latter case?
\end{exe}

For $A = aX$ and $B = bZ$, expand $e^{-i(A+B)} - e^{-iA}e^{-iB}$ to second order in $a$ and $b$, and compare its norm with the bound $\frac12\|[A,B]\| = |ab|$ of \ona{Lemma~\ref{lem:ch05:trotter}}\onp{the Trotter lemma}.
