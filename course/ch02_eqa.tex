\chapter{The Exponential Quantum Advantage Hypothesis}\label{C.EQA}

\ona{Much of the early enthusiasm for quantum computing in chemistry, and with it for the variational methods of this course, rested on the belief that a quantum computer would find the lowest energy of a molecule exponentially faster than any classical one. In this \chap{} we examine that belief. The reader is not expected to know any physical chemistry. We begin by describing, in the language of linear algebra, what it is that chemists compute and why it is difficult. We then describe the quantum algorithm on which the belief rests, quantum phase estimation, and what it demands as molecules become larger. Finally we follow the critical assessment of \citet{lee2023evaluating}, and set it beside the complexity-theoretic view taken in the survey \citep{abbas2024challenges} and beside the exponential speedup by annealing of \citet{nagaj2012quantum}, which is proved in \Chap~\chref{C.Annealing}. Much the same considerations apply to optimization. Worst-case complexity is neither necessary nor sufficient for a practical advantage, and a great deal turns on the quality of the state from which one starts.}

\section{The motivation}

\begin{frame}\ft{Feynman's challenge}
\ona{In May 1981, at a conference on the physics of computation held at MIT, Richard Feynman asked whether physics could be simulated efficiently by a classical computer. For quantum systems he thought not. To describe $n$ two-level systems one has to store $2^n$ complex numbers, and one cannot get round this by working with classical probabilities instead, since quantum mechanics produces ``probabilities'' which may be negative. He suggested building a computer out of quantum-mechanical elements, and closed the lecture, which was published the following year \citep{feynman1982simulating}, with a sentence that has since opened a great many talks:}
\begin{center}
\begin{minipage}{0.86\linewidth}
\textit{``\ldots\ nature isn't classical, dammit, and if you want to make a simulation of nature, you'd better make it quantum mechanical, and by golly it's a wonderful problem, because it doesn't look so easy.''}\\[0.2em]
\hspace*{\fill}--- R.~P.~Feynman, \emph{Simulating physics with computers}, 1982 \citep{feynman1982simulating}
\end{minipage}
\end{center}
\ona{Feynman's argument concerned the size of the \emph{description}, and that is indeed exponential. An algorithm came later. \citet{lloyd1996universal} showed that a quantum computer can simulate efficiently the time evolution of any system with local interactions, and \citet{aspuru2005simulated} proposed that molecular energies be obtained by combining such a simulation with the phase-estimation algorithm of \citet{kitaev1995quantum}. It was then widely supposed that quantum computers would compute molecular energies exponentially faster than classical ones. Feynman himself did \emph{not} claim that the lowest energy is easy to find on a quantum computer, and, as we shall see, in the worst case it is not.}
\onp{\begin{itemize}\item Describing $n$ quantum two-level systems takes $2^n$ numbers. \item Lloyd 1996: time evolution can be simulated efficiently; Aspuru-Guzik et al.\ 2005: molecular energies by phase estimation. \item Feynman did \emph{not} claim that ground states are easy to find.\end{itemize}}
\end{frame}

\section{Quantum chemistry for non-chemists}

\begin{frame}\ft{What chemists compute, and why}
\ona{A molecule consists of atomic nuclei, which are heavy and positively charged, and electrons, which are light and negatively charged. The nuclei are some thousands of times heavier than the electrons, and so one may fix their positions $\boldsymbol R$ and treat only the electrons quantum mechanically; this is the \emph{Born--Oppenheimer approximation} of \Chap~\chref{C.Motivation}. The electrons then have a lowest possible energy $E_0(\boldsymbol R)$, the \emph{ground-state energy}, and most of what a chemist wishes to know can be read off from this function of the nuclear positions:}
\begin{itemize}\setlength\itemsep{0.2em}
    \item \textbf{Structure:} the positions $\boldsymbol R$ at which $E_0(\boldsymbol R)$ is smallest give the shape of the molecule.
    \item \textbf{Reactions:} the difference in $E_0$ between reactants and products tells one whether a reaction releases energy, and the height of the lowest ``mountain pass'' between them (the \emph{barrier}) how quickly it proceeds.
    \item \textbf{Binding:} the energy of a drug bound to a protein, less the energies of the two apart.
\end{itemize}
\ona{The natural unit of energy is the \emph{Hartree} ($1\,\mathrm{Ha} \approx 27.2$ eV $\approx 627.5$ kcal/mol). Reaction rates depend exponentially on barriers, as $e^{-\Delta E/k_BT}$, and chemists therefore ask for \emph{chemical accuracy}, $1$ kcal/mol $\approx 1.6\times10^{-3}$ Ha, which at room temperature determines a rate to within a factor of about five. Two features of this target will matter later. It is an \emph{absolute} error, and it is demanding even for a small and well-behaved molecule. \citet{eriksen2020ground} organised a blind challenge to compute the ground-state energy of benzene, C$_6$H$_6$, in the standard cc-pVDZ basis, with the six innermost (core) orbitals frozen. That is $30$ electrons in $108$ orbitals, and a state space of the order of $10^{35}$ determinants. The \emph{correlation energy}, the part of the energy not captured by the single-determinant Hartree--Fock approximation, is about $-863\times10^{-3}$ Ha, or some $540$ times chemical accuracy. The coupled-cluster hierarchy gives $-862.37\times10^{-3}$ Ha at the highest level that can be afforded (CCSDTQ), and adding triple excitations to CCSD lowers the energy by $36.45\times10^{-3}$ Ha on its own. Even so, the best available methods, some of them run for $1.7$ million core hours, differ with a standard deviation of more than $1.3\times10^{-3}$ Ha. That is comparable with chemical accuracy itself, for a molecule without strong correlation. The second feature is that the energy is \emph{extensive}: it grows in proportion to the number of atoms, so that the relative precision required becomes finer as the molecule grows.}
\onp{\footnotesize Chemical accuracy: $1$ kcal/mol $= 1.6\times10^{-3}$ Ha, an \emph{absolute} error. Benzene, cc-pVDZ \citep{eriksen2020ground}: $30$ electrons in $108$ orbitals, $\sim10^{35}$ determinants; correlation energy $\approx -0.863$ Ha; the best methods differ by about $1.3\times10^{-3}$ Ha, i.e., by about chemical accuracy. The energy grows with the number of atoms.}
\end{frame}

\begin{frame}\ft{From a molecule to a matrix}
\ona{As in \Chap~\chref{C.Motivation}, the space in which the electrons move is discretised by choosing a finite set of functions, the \emph{spin orbitals}, whose number $L$ is proportional to the number of atoms. In the language of quantum computing:}
\begin{itemize}\setlength\itemsep{0.25em}\onp{\setlength\itemsep{0.1em}}
    \item \textbf{State space.} Each spin orbital is either empty or holds one electron. A \emph{configuration} of the electrons is thus a bit string of length $L$, and a general state is a unit vector in $\C^{2^L}$, with one qubit per spin orbital. When the number $N$ of electrons is fixed, only the $\binom{L}{N}$ strings with $N$ ones are needed; chemists call these basis states \emph{Slater determinants}, or simply \emph{determinants}.
    \item \textbf{Hamiltonian.} The energy is a Hermitian matrix $H$ on this space, the \emph{electronic Hamiltonian}. Its entries are integrals over the orbitals, which can be computed classically in polynomial time, and it is a sum of $\mathcal O(L^4)$ terms, each of which moves at most two electrons (equivalently, a sum of $\mathcal O(L^4)$ Pauli strings; \Chap~\chref{C.VQE}).
    \item \textbf{Problem.} Find the smallest eigenvalue $E_0$ of $H$ to precision $\epsilon$. This is the \emph{ground-state energy problem}, and the corresponding eigenvector $\ket{\Psi_0}$ is the \emph{ground state}.
\end{itemize}
\ona{The matrix is exponentially large but has a short description, as did the MaxCut Hamiltonian $H_C$ of \Chap~\chref{C.Motivation}. There, however, $H_C$ was diagonal and the eigenvector a bit string; here $H$ is not diagonal, and the eigenvector is a genuine superposition. Some sizes may help. The nitrogen molecule N$_2$ in a modest basis has $L = 56$ and $N = 14$, and hence $\binom{28}{7}^2 \approx 1.4\times10^{12}$ determinants of the right spin. An \emph{active space}, which is the chemists' term for the subset of orbitals treated accurately while the rest are frozen, of $50$ electrons in $50$ orbitals already has $\binom{50}{25}^2 \approx 1.6\times10^{28}$. Exact diagonalisation, known to chemists as \emph{full configuration interaction} (FCI), is feasible up to about $10^{10}$ determinants.}
\onp{\footnotesize $L$ spin orbitals $=$ $L$ qubits; $\binom{L}{N}$ determinants (bit strings with $N$ ones); $H$ is a sum of $\mathcal O(L^4)$ terms. N$_2$: $1.4\times10^{12}$ determinants; $50$ electrons in $50$ orbitals: $1.6\times10^{28}$. Exact diagonalisation (FCI): up to $\sim10^{10}$.}
\end{frame}

\begin{frame}\ft{Weak and strong correlation}
\ona{The ground state is a vector $\ket{\Psi_0} = \sum_x c_x \ket x$ over bit strings $x$. Chemistry is tractable at all only because, for most molecules, this vector is dominated by a single bit string:}
\onp{\footnotesize}
\begin{itemize}\setlength\itemsep{0.25em}
    \item \textbf{Hartree--Fock (HF).} The orbitals are chosen so that the best single determinant has the lowest possible energy.\ona{ This is a \emph{mean-field} method, in which each electron sees only the average of the others, and it costs $\mathcal O(L^3)$ to $\mathcal O(L^4)$. The HF state plays the part of a product state, and the difference between its energy and $E_0$ is the \emph{correlation energy}. This is usually less than one per cent of the total, but much larger than chemical accuracy (for H$_2$ in \Chap~\chref{C.Motivation} it was $0.021$ Ha).}\onp{ Mean field; plays the part of a product state.}
    \item \textbf{Weakly correlated} molecules, which make up most of organic chemistry, have $|c_{\rm HF}|^2$ close to one\ona{, and the remaining amplitudes are small corrections which can be computed by expanding about HF}.
    \item \textbf{Strongly correlated} systems have many bit strings of comparable weight\ona{, and no expansion about any one of them converges}. \ona{The usual causes are the breaking of chemical bonds (H$_2$ at large $R$ in \Chap~\chref{C.Motivation}) and \emph{transition metals} such as iron, whose partly filled electron shells admit many configurations of nearly equal energy. Several iron atoms coupled together, as in the iron--sulphur clusters below, are the standard example of a hard case.}\onp{Bond breaking; transition metals such as iron.}
\end{itemize}
\ona{A computer scientist may think of HF as standing to the ground state roughly as the best single cut stands to a probability distribution over cuts. The quantity on which most of this \chap{} turns is the \emph{overlap} $S = |\braket{\Phi|\Psi_0}|$ of a cheap starting state $\Phi$, such as HF, with the true ground state. Chemists call $S^2$ the \emph{weight} of $\Phi$.}
\onp{\footnotesize Overlap $S = |\braket{\Phi|\Psi_0}|$ of a cheap state $\Phi$ (e.g.\ HF) with the ground state; $S^2$ is its \emph{weight}. Most of this lecture turns on it.}
\end{frame}

\begin{frame}\ft{A glossary of the classical methods}
\ona{The following names appear in the figures of \citet{lee2023evaluating}. None of them need be understood in detail. They are all \emph{heuristics}, that is, polynomial-time algorithms whose accuracy is not guaranteed but can be checked empirically.}
\onp{\scriptsize}
\begin{itemize}\setlength\itemsep{0.15em}
    \item \textbf{Coupled cluster}, e.g.\ \textbf{CCSD(T)}: writes $\ket{\Psi_0} \approx e^{T}\ket{\rm HF}$, where $T$ is a sum of single and double moves of electrons with unknown coefficients (S, D), and adds a perturbative estimate of triple moves, (T). It costs $\mathcal O(L^7)$ and is regarded as ``the gold standard'' for weakly correlated molecules. \emph{Local} variants use the fact that distant electrons hardly interact, and cost $\mathcal O(L)$ for large molecules.
    \item \textbf{Size-extensive}: the error grows only linearly with size, i.e., it is constant per atom. Coupled cluster has this property; truncated expansions in bit strings do not.
    \item \textbf{DMRG} (density matrix renormalisation group): minimises the energy over \emph{matrix product states}, vectors whose amplitudes are products of $D\times D$ matrices, one for each orbital (\Chap~\chref{C.PerIteration}). The \emph{bond dimension} $D$ controls the accuracy, and the cost is $\mathrm{poly}(L)\,D^3$. It does best when the correlation is chain-like, and is used for active spaces of about $100$ orbitals.
    \item \textbf{PEPS} (projected entangled pair states): the same idea on two- and three-dimensional lattices.
    \item \textbf{Quantum Monte Carlo}, e.g.\ AFQMC: estimates $E_0$ by random sampling. It would be exact but for the \emph{sign problem}, which it handles by an approximation.
    \item \textbf{Configuration state function} (CSF): a fixed combination of a few determinants with the correct total spin; as a cheap starting state it is slightly better than a single determinant.
    \item \textbf{Model Hamiltonians}: the \emph{Heisenberg model}, with one spin at each lattice site and $H = J\sum_{\langle ij\rangle}(X_iX_j + Y_iY_j + Z_iZ_j)$; and the \emph{Hubbard model}, in which electrons hop between neighbouring sites and pay an energy $U$ when two occupy the same site. \emph{Half filling} means one electron per site, and \emph{$1/8$ doping} means one in eight removed, the regime associated with high-temperature superconductivity.
\end{itemize}
\end{frame}

\begin{frame}\ft{The benchmark systems: iron--sulphur clusters}
\ona{The molecules most often cited in support of quantum computing for chemistry are the \emph{iron--sulphur clusters}, small groups of iron and sulphur atoms held within proteins, which carry electrons in photosynthesis, respiration and nitrogen fixation. The largest of them, the iron-molybdenum cofactor \emph{FeMo-co}, with seven iron atoms and one of molybdenum, is the active site of the enzyme nitrogenase, which converts nitrogen from the air into ammonia at room temperature. The industrial equivalent, the Haber--Bosch process, requires high temperature and pressure, and is often said to consume one to two per cent of the world's energy. An understanding of how FeMo-co works would therefore be of real value, and \citet{reiher2017elucidating} proposed it as the leading application of fault-tolerant quantum computing. The smaller clusters, [2Fe-2S] and [4Fe-4S], with two and four iron atoms, serve as the ``hydrogen atoms'' of the subject: small enough to be computed classically, yet correlated enough to be difficult.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item Each iron atom brings several nearly degenerate configurations, and the spins of the iron atoms are coupled, so the correlation is strong and grows with the number of metal centres.
    \item Active spaces range from about $40$ qubits for [2Fe-2S] to $154$ qubits for FeMo-co in the models of \citet{lee2023evaluating}.
\end{itemize}
\ona{Their structures are shown in Fig.~\ref{fig:ch02:eqa1}(A) below.}
\end{frame}

\begin{frame}\ft{How hard is the ground-state problem, formally?}
\ona{For the worst case, complexity theory gives a clear answer. The \emph{local Hamiltonian problem} is as follows: given a Hermitian matrix $H = \sum_j H_j$ on $n$ qubits, with $\mathrm{poly}(n)$ terms each acting on $k$ qubits, and two numbers $a < b$ with $b - a \ge 1/\mathrm{poly}(n)$, decide whether $E_0 \le a$ or $E_0 \ge b$. It is the quantum counterpart of satisfiability:}
\begin{itemize}\setlength\itemsep{0.25em}
    \item It is \textbf{$\mathbf{QMA}$-complete}, already for $k=2$ \citep{kitaev2002classical,kempe2006complexity}. $\mathbf{QMA}$ is the quantum counterpart of $\mathbf{NP}$: the problems whose ``yes'' answers can be checked efficiently by a quantum computer given a quantum state as certificate, here the ground state itself.
    \item The same is true of the electronic Hamiltonians of chemistry with general coefficients \citep{ogorman2022intractability}.
    \item Just as $\mathbf{NP}$-hard problems are not thought to be solvable in polynomial time on a classical computer, $\mathbf{QMA}$-hard problems are not thought to be solvable in polynomial time on a quantum one.
\end{itemize}
\ona{Quantum computers, then, do not solve ground-state problems efficiently in general, and any exponential advantage would have to come from some structure in the instances that arise in chemistry. We shall see that, in the quantum algorithm, the difficulty reappears in the overlap $S$ of the starting state.}
\onp{\footnotesize In the worst case quantum computers are no help either; any advantage must come from the structure of chemical instances.}
\end{frame}

\section{Quantum phase estimation}

\begin{frame}\ft{Energies as phases}
\ona{A quantum computer does not diagonalise matrices; it applies unitaries. The connection is made through \emph{time evolution}. For a time step $\tau$, the unitary $U = e^{-iH\tau}$ has the same eigenvectors as $H$, and if $H\ket{\Psi_k} = E_k\ket{\Psi_k}$ then}
\begin{equation}\label{eq:ch02:phase}
    U\ket{\Psi_k} = e^{-iE_k\tau}\ket{\Psi_k} = e^{2\pi i\varphi_k}\ket{\Psi_k}, \qquad \varphi_k = -\frac{E_k\tau}{2\pi} \bmod 1.
\end{equation}
\ona{To measure an energy is therefore to measure the \emph{phase} $\varphi_k \in [0,1)$ of an eigenvalue of a unitary. If the phases are not to be ambiguous modulo $1$, the step $\tau$ must be small enough for the whole spectrum to fit within one period, $\tau \le 2\pi/(E_{\max} - E_{\min})$; after shifting $H$ by a multiple of the identity one may take $\tau \approx \pi/\|H\|$. The controlled version of $U$ can be implemented with $\mathrm{poly}(L)$ gates, by Trotter product formulae or by more refined methods \citep{lloyd1996universal,nielsen2002quantum}.}
\ona{The basic building block is the \emph{Hadamard test}, which we met in \Chap~\chref{C.Motivation}. An extra qubit, the \emph{ancilla}, is prepared in $\ket{+} = (\ket0+\ket1)/\sqrt2$; $U^{t}$ is applied to the system, controlled on the ancilla; and the ancilla is measured in the $\ket\pm$ basis. On an eigenstate, the controlled-$U^t$ merely multiplies the $\ket1$ branch by $e^{2\pi i t\varphi_k}$ (\emph{phase kickback}), and}
\begin{equation}\label{eq:ch02:hadamardtest}
    \mathrm{Pr}[\text{outcome } +] = \tfrac12\big(1 + \cos(2\pi t\varphi_k)\big).
\end{equation}
\ona{Repeating this for a few values of the integer power $t$ determines $\varphi_k$, and with it $E_k$.}
\onp{\footnotesize $U = e^{-iH\tau}$: the energy $E_k$ becomes a phase $\varphi_k$. Hadamard test with controlled-$U^t$: $\mathrm{Pr}[+] = \frac12(1+\cos 2\pi t\varphi_k)$.}
\end{frame}

\begin{frame}\ft{The phase-estimation circuit}
\ona{Textbook quantum phase estimation (QPE) \citep{kitaev1995quantum,nielsen2002quantum} reads off $m$ binary digits of $\varphi_k$ together. Ancilla $j$ controls $U^{2^{j}}$, $j = 0, \dots, m-1$, and so acquires the phase $2^j\varphi_k$; the inverse quantum Fourier transform then turns these phases into the binary expansion of $\varphi_k$, which is measured:}
\ona{\begin{figure}[htb]}\begin{center}

\ona{\caption{Quantum phase estimation with $m = 3$ ancillas. Ancilla $j$ controls $U^{2^j}$, and the inverse quantum Fourier transform maps the phases kicked back onto the ancillas to the binary digits of $\varphi$. The system register is left approximately in the eigenstate whose phase was measured.}\label{fig:ch02:qpe}\end{figure}}
\begin{itemize}\setlength\itemsep{0.2em}
    \item \textbf{Eigenstate input:} the output is the best $m$-bit approximation of $\varphi_k$ with probability at least $4/\pi^2$; with $m = b + \lceil\log_2(2 + \frac1{2\delta})\rceil$ ancillas it is correct to $b$ bits with probability at least $1-\delta$ \citep{nielsen2002quantum}.
    \item \textbf{General input} $\ket\Phi = \sum_k c_k\ket{\Psi_k}$: by linearity the output approximates $\varphi_k$ with probability $|c_k|^2$, and the system register is left approximately in $\ket{\Psi_k}$. The ground-state energy is obtained with probability $|c_0|^2 = S^2$.
    \item \textbf{Cost:} $2^m - 1$ applications of controlled-$U$, a total evolution time of $T = (2^m-1)\tau$.
\end{itemize}
\ona{A variant with a single ancilla, which obtains one digit at a time from the Hadamard test \eqref{eq:ch02:hadamardtest}, has the same total cost \citep{kitaev1995quantum}. The post-QPE methods of \citet{lt21} and \citet{zhang2022computing}, described in \Chap~\chref{C.Motivation}, replace the Fourier transform by classical post-processing and use shorter circuits.}
\end{frame}

\begin{frame}\ft{The Heisenberg limit}
\ona{An energy precision $\epsilon$ corresponds to a phase precision $\epsilon\tau/2\pi$, that is, to $b = \log_2(2\pi/(\epsilon\tau))$ bits. The total evolution time is therefore}
\begin{equation}\label{eq:ch02:heisenberg}
    T \;\approx\; 2^b\,\tau \;=\; \frac{2\pi}{\epsilon},
\end{equation}
\ona{multiplied by $\mathcal O(1/\delta)$ for a failure probability $\delta$, or by $\mathcal O(\log 1/\delta)$ if one repeats and takes the median. This $1/\epsilon$ scaling is known as the \emph{Heisenberg limit}, and it is what makes QPE attractive. A method which estimates $\braket{\Phi|H|\Phi}$, or a single interference probability, by sampling needs $\mathcal O(1/\epsilon^2)$ repetitions by the central limit theorem; the variational methods of \Chap~\chref{C.VQE} are of this kind. More importantly, such a method measures the energy of the state one happens to have, and not an eigenvalue. QPE measures an eigenvalue, whatever the input state, provided one is prepared to repeat it until the right eigenvalue appears.}
\onp{\begin{itemize}\item Total evolution time $T \approx 2\pi/\epsilon$: the \emph{Heisenberg limit}. \item Sampling an expectation value (as in VQE) costs $\mathcal O(1/\epsilon^2)$, and gives the energy of the state one has, not an eigenvalue.\end{itemize}}
\ona{The number of gates is the evolution time, in units of $\tau$, multiplied by the cost of one controlled-$U$:}
\begin{equation}\label{eq:ch02:gates}
    G \;\approx\; \underbrace{\frac{2\pi}{\epsilon\,\tau}}_{\approx\,2\|H\|/\epsilon}\;\times\;\underbrace{\mathrm{poly}(L)}_{\text{one step of }U}\;=\;\mathrm{poly}(L)\,\frac{\|H\|}{\epsilon}.
\end{equation}
\ona{In the most efficient methods, based on \emph{qubitization}, $\|H\|$ is replaced by $\lambda = \sum_j|h_j|$, the sum of the absolute values of the coefficients of $H = \sum_j h_jP_j$ in the Pauli basis; this too is $\mathrm{poly}(L)$.}
\end{frame}

\begin{frame}\ft{What QPE requires as the molecule grows}
\ona{These formulae do not show how the accuracy required depends on the size $L$ and on the dimension $2^L$ of the state space. We take the requirements in turn.}
\onp{\begin{center}\scriptsize
\begin{tabular}{lll}\toprule
requirement & grows with $L$ as & \\\midrule
phase bits $b$ & $\log_2(\|H\|/\epsilon)$ & cheap \\
depth, evolution time & $\|H\|/\epsilon = \mathrm{poly}(L)$ & absolute $\epsilon$, $\|H\|\propto L^{\ge1}$ \\
eigenvalue spacing & $\mathrm{poly}(L)\,2^{-L}$ & need $\epsilon \lesssim \Delta$ to project \\
leakage of tails & $+\,\mathcal O(\log 1/S^2)$ bits & many levels near $E_0$ \\
repetitions & $1/S^2$ & \textbf{exponential if $S$ decays} \\
Trotter steps, coefficients & $\mathrm{poly}(L)$; $\epsilon/L^4$ each & implementation \\
gate error & $\ll 1/G$, $G = \mathrm{poly}(L)/\epsilon$ & error correction \\\bottomrule
\end{tabular}\end{center}}
\ona{\begin{enumerate}\setlength\itemsep{0.25em}
    \item \textbf{Absolute and relative precision.} Chemical accuracy is an absolute $\epsilon$, whereas $|E_0|$ and $\|H\|$ grow at least linearly in $L$. The relative precision $\epsilon/\|H\|$ therefore falls as $1/\mathrm{poly}(L)$. This needs only $b = \log_2(\|H\|/\epsilon) + \mathcal O(1)$ bits, which is cheap, but a circuit depth of $\|H\|/\epsilon$, which grows polynomially and is not. If a fixed error \emph{per atom} $\bar\epsilon$ is enough, then $\epsilon = \bar\epsilon L$ and the depth grows more slowly.
    \item \textbf{An exponentially dense spectrum.} The $2^L$ eigenvalues lie in an interval of width $2\|H\| = \mathrm{poly}(L)$, and so are spaced on average $\mathrm{poly}(L)\,2^{-L}$ apart. To resolve a typical eigenvalue would need exponential precision, and so exponential time. QPE succeeds for the ground state only because it lies at the edge of the spectrum, separated from the rest by the \emph{gap} $\Delta = E_1 - E_0$. To project onto $\ket{\Psi_0}$, rather than merely to estimate $E_0$, one needs $\epsilon \lesssim \Delta$. In gapped molecules $\Delta$ remains bounded, but it may close as $1/\mathrm{poly}(L)$ or faster, as in metals, in critical systems, or among the nearly degenerate spin states of iron clusters.
    \item \textbf{Leakage from the rest of the spectrum.} For a given eigenphase, the output distribution of QPE has tails which decay only as the inverse square of the distance. If $S^2$ is small, the other eigenstates carry nearly all the weight, $1-S^2$, and their tails may give outcomes near $E_0$, or even below it, more often than the ground state does. Pushing them below $S^2$ requires additional bits, or tapered variants of QPE, at a cost growing with $\log(1/S^2)$. It also means that simply taking the smallest outcome gives a biased estimate.
    \item \textbf{Repetitions.} QPE returns $E_0$ with probability $S^2$, so one expects $1/S^2$ runs, or $1/S$ with amplitude amplification. If $S$ decays exponentially in $L$, so does the running time, however cheap the circuit.
    \item \textbf{Accuracy of the implementation.} Approximating $e^{-iH\tau}$ by products of simpler exponentials (Trotterisation) introduces an error in the energy, which must be kept below $\epsilon$; the number of Trotter steps grows polynomially in $L$ and in $1/\epsilon$. The $\mathcal O(L^4)$ coefficients of $H$, which are computed classically, must each be accurate to about $\epsilon/L^4$, since their errors accumulate in $\|H\|$.
    \item \textbf{Accuracy of the hardware.} A circuit of $G$ gates succeeds only if each gate fails with probability well below $1/G$. With $G \sim \mathrm{poly}(L)/\epsilon$ running into the billions, and physical error rates of about $10^{-3}$, one needs quantum error correction with logical error rates of $10^{-10}$ or less, and the number of physical qubits per logical qubit grows as $\log^2 G$. That is why QPE is described as a \emph{fault-tolerant} algorithm.
\end{enumerate}}
\ona{Items 1, 2, 5 and 6 are polynomial or polylogarithmic in $L$: expensive, but not exponential. It is in item 4 that an exponential can enter, and the remainder of this \chap{} is concerned with it.}
\end{frame}

\begin{frame}\ft{The cost of the fault-tolerant algorithm}
\ona{We are now in a position to follow the accounting of \citet{lee2023evaluating}. The cost of QPE to precision $\epsilon$ has three parts: the preparation of a trial state $\Phi$, at cost $C$; the phase-estimation circuit itself; and the repetitions needed before the ground state $\Psi_0$, rather than some other eigenstate, is obtained:}
\begin{equation}\label{eq:ch02:qpecost}
    \mathrm{poly}(1/S)\,\big[\mathrm{poly}(L)\,\mathrm{poly}(1/\epsilon) + C\big],\qquad S = |\braket{\Phi|\Psi_0}|,
\end{equation}
\ona{with $\mathrm{poly}(1/S) = 1/S^2$ for QPE.}
\ona{The informal argument for an exponential advantage cites the $\mathrm{poly}(L)$ of the circuit and assumes that the classical cost is $\exp(L)$, the cost of exact diagonalisation. The number of repetitions, however, also depends on $L$, through the quality of the state preparation. Worst-case $\mathbf{QMA}$-hardness is consistent with \eqref{eq:ch02:qpecost}: on hard instances, no efficiently preparable $\Phi$ has $S \ge 1/\mathrm{poly}(L)$. To restrict attention to ``generic'' chemistry is, in effect, to assume that some unspecified structure prevents good state preparation from being exponentially hard. But the same structure might equally assist the classical heuristics, and for an exponential advantage the state preparation must be exponentially cheaper than the classical solution. Post-QPE algorithms of lower depth, such as the Fourier methods of \Chap~\chref{C.Motivation}, alter the constants and the dependence on $\epsilon$, but retain the factor $1/S^2$ \citep{lt21,zhang2022computing}.}
\onp{The $\mathrm{poly}(L)$ is the circuit; the $\mathrm{poly}(1/S)$ is the number of repetitions, and depends on $L$ through the trial state. $\mathbf{QMA}$-hardness reappears in $S$. Post-QPE methods retain the $1/S^2$ \citep{lt21,zhang2022computing}.}
\end{frame}

\section{The hypothesis}

\begin{frame}\ft{What is claimed}
\ona{It is often said, or implied, that quantum computers offer an exponential advantage for the ground-state energy problem. \citet{lee2023evaluating} state the claim carefully, and then weigh the evidence for it:}
\begin{center}
\emph{The exponential quantum advantage (EQA) hypothesis:} for a large set of relevant (``generic'') chemical problems, the ground-state energy can be computed exponentially faster, as a function of system size, on a quantum than on a classical computer.
\end{center}
\ona{Since the ground states of even simple Hamiltonians are $\mathbf{QMA}$-hard to compute, the hypothesis cannot concern the worst case. What it asserts is that \emph{generic} chemistry involves Hamiltonians which are (i) polynomially easy for quantum algorithms, including the preparation of the ground state, and at the same time (ii) exponentially hard classically, even for the best classical heuristics. Both conditions must hold on the \emph{same} instances. Neither can be proved or disproved, because ``generic chemistry'' is not a well-defined set; the question is whether EQA should be thought likely in the cases of interest, or whether it requires very particular circumstances. \citet{lee2023evaluating} confine themselves to fault-tolerant algorithms, the setting most favourable to quantum computing, in which there is no noise.}
\onp{\begin{itemize}\item EQA: generic chemistry is (i) polynomially easy for a quantum computer, including state preparation, \emph{and} (ii) exponentially hard for every classical heuristic, on the same instances. \item It cannot be proved or disproved (``generic'' is not defined); the question is whether it is likely, or requires very particular circumstances \citep{lee2023evaluating}.\end{itemize}}
\end{frame}

\section{Weighing the evidence}

\begin{frame}\ft{Half (i): ansatz state preparation and the orthogonality catastrophe}
\ona{The simplest way of obtaining $\Phi$ is to prepare a state given by a classical \emph{ansatz}, an assumed functional form, such as the Hartree--Fock determinant, which becomes a single bit string after a change of orbital basis, and to hope that its overlap is good. For small molecules it is. EQA, however, concerns large $L$, and there a simple argument, the \emph{orthogonality catastrophe}, gives cause for concern. For $K$ weakly interacting subsystems the ground state is approximately a product, and if the ansatz has overlap $s < 1$ with each local ground state, the overall overlap is $s^K$, which is exponentially small. With $s^2 = 0.99$ for each subsystem, for example, $S^2$ falls below $10^{-6}$ once $K > 1375$.}
\ona{This does not in itself rule out preparation from an ansatz. Neither the ansatz nor the true state need be of product form, and the local overlaps may be improved as $L$ grows so as to reach $1/\mathrm{poly}(L)$ overall. The point that matters for EQA is another one: if classical heuristics can efficiently prepare states of such good overlap, they may well be able to obtain the energy efficiently to the required precision too. \citet{lee2023evaluating} examine this for the iron--sulphur clusters, with up to 154 qubits, taking DMRG as the reference for $\Psi_0$:}
\figslide[height=0.4\textheight]{figures/eqa/fig1.pdf}{(A) The iron--sulphur clusters [2Fe-2S], [4Fe-4S], the P-cluster and FeMo-co (iron orange, sulphur yellow). (B) Weight $S^2$ of the best single determinant (purple) and of the best configuration state function (orange) in the ground state, against the number of metal centres. The decay is exponential, reaching about $10^{-7}$ for FeMo-co \citep{lee2023evaluating}.}{fig:ch02:eqa1}
\ona{Is the decay unavoidable? For the [2Fe-2S] fragment, improving the local overlap to an infidelity $I$ (that is, $1 - s^2 = I$) by a classical method costs $\mathrm{poly}(1/I)$, but the energy error of that same method is by then already $\mathrm{poly}(I)$. Improving the overlap sufficiently for efficient state preparation thus also solves the problem classically. EQA would need a heuristic which efficiently gives an overlap of $1/\mathrm{poly}(L)$ and yet cannot reach the required precision in the energy, and no such heuristic is known.}
\onp{\footnotesize Local overlap $s$ $\Rightarrow$ overall overlap $s^K$. Improving it classically also solves the problem classically.}
\end{frame}

\begin{frame}\ft{Half (i), continued: adiabatic state preparation}
\ona{Alternatively, $\Phi$ may be prepared on the quantum computer itself by \emph{adiabatic state preparation} (ASP; \Chap~\chref{C.Annealing}). One starts in the ground state of a simple Hamiltonian $H(0)$, for instance one whose ground state is HF, and changes the Hamiltonian slowly along a path $H(s)$, $s \in [0,1]$, to the target $H(1) = H$. The adiabatic theorem guarantees that the state follows the instantaneous ground state, provided the change is slow compared with the gap $\Delta(s)$ between the two lowest eigenvalues along the path. If the minimum gap is at least $1/\mathrm{poly}(L)$ (a \emph{protected} gap), ASP followed by QPE is polynomial. Nothing guarantees such a gap for a given path, however, and a closing gap is just how $\mathbf{QMA}$-hardness would show itself; ASP is therefore also a heuristic. For a 24-qubit model of [2Fe-2S], \citet{lee2023evaluating} estimate the time required as $T_{\rm ASP} \sim \max_s|\braket{\Upsilon_0(s)|dH/ds|\Upsilon_1(s)}|/\Delta(s)^2$, where $\Upsilon_0(s)$ and $\Upsilon_1(s)$ are the two lowest eigenvectors of $H(s)$. They check this estimate against simulations of the time evolution, for two families of initial Hamiltonians:}
\figslide[height=0.4\textheight]{figures/eqa/fig2.pdf}{Adiabatic state preparation for a [2Fe-2S] model \citep{lee2023evaluating}. (A) Structure and active space. (B) The adiabatic estimate agrees with simulated ASP times to within $O(1)$. (C) Estimated ASP time, for mean-field and for interacting initial Hamiltonians, against the weight of the initial ground state in the final one. It spans eight orders of magnitude and scales as $\mathrm{poly}(1/|\braket{\Upsilon_0|\Psi_0}|)$, rather as in unstructured search.}{fig:ch02:eqa2}
\ona{Starting from the lowest-energy mean-field state gives $T_{\rm ASP} > T_{\rm QPE}$: preparing the state takes longer than running phase estimation on it. Among the interacting starting points, 20 of the 24 qubits must be included in the exactly diagonalised initial Hamiltonian before $T_{\rm ASP} < T_{\rm QPE}$. Once again a good starting point comes from classical heuristics, which would then have to be exponentially better at starting than at solving.}
\onp{\footnotesize ASP time $\sim\mathrm{poly}(1/S)$ of the initial state: again the overlap matters most.}
\end{frame}

\begin{frame}\ft{Annealing can be exponentially faster: Somma, Nagaj, Kieferov\'a}
\ona{The estimate $T_{\rm ASP} \sim 1/\Delta^2$ used above is a \emph{sufficient} condition for adiabatic evolution, but not a necessary one. That the two can differ exponentially was shown by \citet{nagaj2012quantum} in \emph{Quantum Speedup by Quantum Annealing}, which we study in full in \Chap~\chref{C.Annealing}:}
\begin{itemize}\setlength\itemsep{0.25em}
    \item In the \emph{glued-trees} problem, an oracle problem on a graph with $2^{\mathcal O(n)}$ vertices, every classical algorithm needs exponentially many queries to the oracle \citep{childs2003exponential}.
    \item Quantum annealing along a suitable path of \emph{stoquastic} Hamiltonians, those without a sign problem and so in a sense the most ``classical'', finds the exit in $\mathrm{poly}(n)$ time: an \emph{exponential} speedup over any classical algorithm.
    \item The minimum gap along the path is nonetheless exponentially small. The schedule does not avoid it but makes use of it: the state passes to the first excited level at one avoided crossing and returns at a symmetric one, while the third level remains polynomially far away.
\end{itemize}
\ona{What follows for EQA? In favour, there is no general obstacle to exponentially fast state preparation on a quantum computer; estimates based on the gap may be much too pessimistic, and diabatic shortcuts can do better. Against, the speedup is relative to an oracle, and it depends on a special symmetry, under which the evolution stays within a subspace of dimension $2n+2$, that has no known analogue in molecules. It shows that exponential speedups in state preparation are possible, but in a very particular setting of the kind that \citet{lee2023evaluating} would call fine-tuned. Whether chemically relevant Hamiltonians possess such structure is an open question.}
\onp{\footnotesize Annealing can be exponentially faster, even with an exponentially small gap, but on an oracle problem with a special symmetry: possible in principle, not shown for generic chemistry (\Chap~\chref{C.Annealing}).}
\end{frame}

\begin{frame}\ft{Half (ii): the scaling of classical heuristics}
\ona{The second half of EQA assumes that the best classical heuristics cost $\exp(L)$ at a fixed error $\epsilon$, or at a fixed error per unit, $\bar\epsilon$. The questions to ask are these. Do the heuristics cost $\exp(L)$ within their domain of applicability? Do the various heuristics, taken together, cover chemical space? And where they do not, are exponential methods needed? Since heuristics run at $\mathrm{poly}(L)$ cost without any guarantee of accuracy, one must also measure how their cost depends on the precision; a cost of $\mathrm{poly}(L)\exp(1/\bar\epsilon)$ would be $\exp(L)$ at a fixed total error $\epsilon = \bar\epsilon L$.}
\figslide[height=0.42\textheight]{figures/eqa/fig3.pdf}{Classical heuristics for molecules \citep{lee2023evaluating}. (A) N$_2$: coupled-cluster error against a measure of time, $\mathrm{poly}(1/\epsilon)$. (B) Local (reduced-scaling) CCSD(T) on $n$-alkanes, chains of carbon atoms with attached hydrogens, up to C$_{120}$H$_{242}$: nearly linear cost and a constant error per carbon atom. (C) A fragment of photosystem II, the protein that splits water in photosynthesis, with thousands of electrons, treated on a few processor cores.}{fig:ch02:eqa3}
\ona{For weakly correlated (``single-reference'') molecules, coupled cluster is size-extensive and its cost is $\mathrm{poly}(1/\epsilon)$ in the precision, while local variants cost $\mathcal O(L)$ on gapped systems; the conjectured complexity is $\mathrm{poly}(L)\,\mathrm{poly}(1/\epsilon)$.}
\onp{\footnotesize Weak correlation: coupled cluster, $\mathrm{poly}(L)\,\mathrm{poly}(1/\epsilon)$, nearly linear when locality is used.}
\end{frame}

\begin{frame}\ft{Half (ii), continued: strongly correlated models}
\ona{For strong correlation, the evidence of \citet{lee2023evaluating} comes from lattice models, namely the Heisenberg and Hubbard models of the glossary, and these are very special cases. Their interactions act only between neighbouring sites of a regular lattice, so they have $\mathcal O(L)$ terms rather than the $\mathcal O(L^4)$ of an ab initio Hamiltonian. They are translation invariant, so they are described by one or two parameters, and a tensor network can repeat the same few tensors across the whole lattice. Their ground states also have little entanglement across any cut, as is believed to be true of such local, regular models. This is the structure for which tensor networks were devised: matrix product states and PEPS are efficient because they give compact representations of low-entanglement states of local lattice Hamiltonians. To choose these models as the test cases for strong correlation therefore flatters the classical methods.}
\figslide[height=0.45\textheight]{figures/eqa/fig4.pdf}{Classical heuristics for correlated materials \citep{lee2023evaluating}. (A) PEPS tensor networks on the 3D Heisenberg cube, from $3^3$ to $10^3$ sites, at bond dimension $D = 4$: constant relative error per site and nearly linear time. (B) The 2D Hubbard model at half filling and at the harder $1/8$ doping, on $4\times4$ to $16\times4$ lattices: $1/\bar\epsilon \sim \mathrm{poly}(D)$, depending only weakly on $L$.}{fig:ch02:eqa4}
\ona{Tensor networks of bond dimension $D$ cost $\mathrm{poly}(L)\,\mathrm{poly}(D)$ per iteration, and with the observed $\bar\epsilon \sim 1/\mathrm{poly}(D)$ the empirical cost is $\mathrm{poly}(L)\,\mathrm{poly}(1/\bar\epsilon)$. This does not carry over to \emph{ab initio} Hamiltonians, in which the Coulomb interaction couples every orbital to every other, there is neither a lattice nor translation invariance, and nothing ensures that the ground state has little entanglement. These data do not tell us how classical heuristics scale for strongly correlated molecules such as the iron--sulphur clusters. Many such problems remain beyond classical reach, and whether the obstacle is polynomial or exponential is not known.}
\onp{\footnotesize The Heisenberg and Hubbard models are very special: local, translation invariant, $\mathcal O(L)$ terms, little entanglement, i.e., the cases tensor networks were devised for. There the cost is empirically $\mathrm{poly}(L)\,\mathrm{poly}(1/\bar\epsilon)$; this flatters the classical methods and says little about ab initio Hamiltonians.}
\end{frame}

\begin{frame}\ft{Conclusions and frequently asked questions}
\ona{\citet{lee2023evaluating} conclude that heuristics are needed for efficient quantum state preparation, and that, since EQA concerns the \emph{ratio} of costs, the same structure that allows classical heuristics to succeed probably also makes state preparation efficient, without giving EQA. Numerical results do not prove asymptotic statements, and EQA for particular problems is not excluded; no relevant chemical system with strong evidence for it has yet been identified. They also answer some frequently asked questions:}
\onp{\footnotesize}
\begin{itemize}\setlength\itemsep{0.2em}
    \item \emph{Does this mean that quantum computers are of no use for ground states?} No. Polynomial advantages, or large constant factors, may be very useful; but the degree of polynomial advantage should be compared with what is available in other applications.
    \item \emph{If classical heuristics fail on a problem and exact methods are exponential, is that EQA?} Not in itself. One must show that reaching the required precision classically \emph{needs} exponential effort, and that the quantum algorithm reaches it without exponential effort, state preparation included.
    \item \emph{What of improvements to quantum algorithms?} They must change the ratio of quantum to classical cost; building locality into a quantum heuristic, for example, usually helps the classical heuristic as well.
    \item \emph{What of other tasks, such as dynamics?} Nothing follows from this analysis, though heuristics deserve careful consideration there too.
\end{itemize}
\end{frame}

\begin{frame}\ft{What can be guaranteed given the overlap}
\ona{The positive results also depend on the overlap. \citet{zhang2022computing} show that if one \emph{is} given a state of overlap $\eta = S^2$, and the Hamiltonian has gap $\gamma$, then not only the ground-state energy \citep{lt21} but any ground-state property $\braket{\psi_0|O|\psi_0}$ can be estimated to accuracy $\epsilon$, with maximal evolution time $\widetilde{\mathcal O}(\gamma^{-1})$, total evolution time $\widetilde{\mathcal O}(\gamma^{-1}\epsilon^{-2}\eta^{-2})$ and a single ancilla; see \Chap~\chref{C.Motivation}. Two remarks may be made:}
\begin{itemize}\setlength\itemsep{0.3em}
    \item The guarantee is conditional on $\eta$, and the cost grows as $\eta^{-2}$, which is the $\mathrm{poly}(1/S)$ of \eqref{eq:ch02:qpecost}. The question raised by EQA is whether $\eta$ remains polynomial along families of instances.
    \item Even under these assumptions the task is not classically easy: estimating ground-state properties classically to high precision would imply $\mathbf P = \mathbf{BQP}$. Without the assumptions, the decision version is $\mathsf{P^{QMA[\log]}}$-complete, and so harder than estimating the energy. One can therefore say what form an advantage would take, namely polynomial cost given a good trial state, on instances where classical heuristics fail to converge. Whether natural families of such instances exist has not been established.
\end{itemize}
\end{frame}

\section{Heuristics, formally, and what this means for optimization}

\begin{frame}\ft{Heuristics, formally}
\ona{A similar difficulty arises in optimization, and the survey \citep{abbas2024challenges} discusses it in some detail. Heuristic algorithms are those without proven guarantees of performance, of running time, or of both. One way of making statements about them exact is to consider distributions of inputs: a pair $(L,D)$, consisting of a language and a family of input distributions, is in $\mathbf{HeurBPP}$ if there is a polynomial-time randomised algorithm $\mathcal A$ such that, for all $n$ and $\delta > 0$,}
\begin{equation*}
    \mathrm{Pr}_{x \sim D_n}\Big[\mathrm{Pr}\big[\mathcal{A}(x,0^{1/\delta}) = L(x)\big] \ge 2/3\Big] \ge 1-\delta.
\end{equation*}
\ona{The algorithm is thus allowed to err on a small fraction of instances. $\mathbf{HeurBQP}$ is the quantum analogue. (Here $L$ denotes a language, and not the number of orbitals.)}
\begin{itemize}\setlength\itemsep{0.2em}
    \item Super-polynomial separations between the two are known only for artificial distributions, essentially images of factoring under Karp reductions, which are of no practical interest.
    \item The overlap gap property makes certain random constraint satisfaction problems hard on average for large classes of classical \emph{and} low-depth quantum algorithms (\Chap~\chref{C.Inapprox}); whether deeper quantum algorithms avoid it is not known.
\end{itemize}
\end{frame}

\begin{frame}\ft{What complexity theory does and does not tell us}
\ona{It is generally believed that quantum computers will not give exponential speedups for $\mathbf{NPO}$-hard problems in the worst case. In practice, however, one seldom needs to solve the worst case, or to solve exactly; approximate solutions will usually do, and the hardness of \emph{approximation} on quantum computers has received little study. The survey's summary applies equally to chemistry and to optimization:}
\begin{center}
    \textit{Understanding complexity theory is extremely useful for gauging a possible quantum advantage, but rigorous complexity-theoretic separations are neither necessary nor sufficient when seeking a practical quantum advantage.}
\end{center}
\ona{The travelling salesman problem is a good example. Its decision version is $\mathbf{NP}$-hard and its optimization version $\mathbf{APX}$-hard; yet Euclidean TSP has a polynomial-time approximation scheme, heuristics solve instances with millions of nodes to near-optimality, and shortest path is in $\mathbf{P}$. On the other hand, some problems with fewer than 100 variables, such as low-autocorrelation binary sequences, are hard classically. The glued trees of \citet{nagaj2012quantum} illustrate the opposite case: a rigorous exponential separation, but for a problem that nobody needs to solve.}
\end{frame}

\begin{frame}\ft{Chemistry and optimization compared}
\ona{The analysis of \citet{lee2023evaluating} carries over, with little change, to the optimization problems of this course:}
\begin{center}\ona{\small}\onp{\scriptsize}
\begin{tabular}{lll}\toprule
 & ground-state chemistry & combinatorial optimization \\\midrule
problem & smallest eigenvalue of $H$ & $\min_x f(x)$, i.e., of diagonal $H_C$ \\
worst case & $\mathbf{QMA}$-hard & $\mathbf{NP}$-hard, often $\mathbf{APX}$-hard \\
trial state & HF, CSF, DMRG state & uniform, warm start (GW cut) \\
overlap $S^2$ & weight of $\Phi$ in $\Psi_0$ & $\mathrm{Pr}[\text{sample is optimal}]$ \\
quantum route & QPE, ASP, VQE & Grover-type search, annealing, QAOA \\
classical baseline & coupled cluster, DMRG, AFQMC & GW; Gurobi, BiqBin, MADAM \\
proven exp.\ speedup & none for generic chemistry & oracle problems only (glued trees) \\\bottomrule
\end{tabular}
\end{center}
\ona{For a diagonal $H_C$, phase estimation on the uniform superposition finds the optimum with probability $S^2 = (\text{number of optima})/2^n$, and amplitude amplification gives Grover's quadratic speedup, $1/S$. This is the orthogonality catastrophe in its simplest form.}
\end{frame}

\begin{frame}\ft{Lessons for variational methods}
\onp{\footnotesize}
\begin{enumerate}\ona{\setlength\itemsep{0.3em}}\onp{\setlength\itemsep{0.1em}}
    \item \textbf{Claims should concern families of instances}, and the classical baseline should be both the best available polynomial-time algorithm and the best classical solvers, not brute force. For MaxCut these are the Goemans--Williamson algorithm (\Chap~\chref{C.Motivation}) and solvers such as Gurobi and the semidefinite branch-and-bound solvers BiqMac\ona{ \citep{rendl2010solving}}, BiqCrunch\ona{ \citep{krislock2017biqcrunch}}, BiqBin\ona{ \citep{gusmeroli2022biqbin}} and MADAM\ona{ \citep{hrga2021madam}}.
    \item \textbf{Much depends on state preparation.} In chemistry this means the overlap $S$; in QAOA, the initial state and the angles. Warm starts (\Chaps~\chref{C.WSRounded}--\chref{C.WSContinuous}) are the counterpart in optimization of a good trial state, and variational methods (\Chap~\chref{C.VQE}) are one means of supplying trial states for QPE.
    \item \textbf{Precision has a price.} QPE attains the Heisenberg limit $1/\epsilon$ but requires fault tolerance; variational methods pay $1/\epsilon^2$ in samples, and measure the energy of the state they have.
    \item \textbf{Estimates based on the gap may be pessimistic.} Annealing and QAOA can make use of small gaps, and not merely survive them (\Chaps~\chref{C.Annealing}--\chref{C.QAOA}).
    \item \textbf{Comparisons should be empirical and fair}: problems that do not depend on a particular model, time to solution and cost, distributions over instances, and validation on hardware (\Chap~\chref{C.PerIteration}).
    \item \textbf{Polynomial advantages are worth having}, and small improvements matter in applications such as finance (\Chap~\chref{C.Portfolio}); but they must be measured, not assumed.
\end{enumerate}
\end{frame}

\begin{frame}\ft{Exercises}
\begin{exe}[Precision of phase estimation]
Consider a molecule with $\|H\| \approx 250$ Ha (a value assumed for the purpose of illustration) and chemical accuracy $\epsilon = 1.6\times10^{-3}$ Ha. (a) How many bits $b$ of the phase are needed? (b) How many applications of controlled-$U$, with $\tau = \pi/\|H\|$, does textbook QPE require, if the bits depending on $\delta$ are ignored? (c) How do the answers change for a molecule ten times as large, if the target is (i) chemical accuracy in total, (ii) chemical accuracy for each unit of the original size?
\end{exe}
\end{frame}

\begin{frame}\ft{Exercises, continued}
\begin{exe}[Orthogonality catastrophe]
A molecule consists of $K$ identical, non-interacting fragments, and the Hartree--Fock state has weight $s^2$ in the ground state of each fragment. (a) Show that the overall weight is $S^2 = s^{2K}$. (b) For $s^2 = 0.99$ and for $s^2 = 0.9$, for which $K$ does the expected number of QPE repetitions exceed $10^6$? (c) Explain why improving $s$ by a classical method may make QPE unnecessary.
\end{exe}
\begin{exe}[Leakage]
For an eigenphase $\varphi$ that is not an $m$-bit number, the probability that textbook QPE outputs the $m$-bit number at distance $j/2^m$ from $\varphi$ is at most $1/(4j^2)$ for $j\ge1$ (up to rounding). Suppose that $S^2 = 10^{-4}$ and that all the remaining weight lies in a single eigenstate $10$ bins above the ground state. Estimate how often QPE returns an outcome at or below the ground-state bin that comes from the wrong eigenstate, and compare this with how often it returns the correct one.
\end{exe}
\end{frame}
